\begin{tcolorbox}[colback=uimmlightblue!40, colframe=uimmblue, boxrule=0.8pt, arc=2pt, left=8pt, right=8pt, top=4pt, bottom=4pt, title={\bfseries Rappel : résoudre $a\,y' + b\,y = c$ ($a$, $b$, $c$ constantes, $a \neq 0$)}, coltitle=white, colbacktitle=uimmblue]
\small
\begin{enumerate}[leftmargin=*, itemsep=1pt, topsep=1pt]
  \item \textbf{Réduire l'équation} : tous les termes en $y'$ et en $y$ à gauche, la constante à droite. Lire $a$, $b$ et $c$.
  \item \textbf{Forme normalisée} : diviser par $a$, on obtient $y' + \alpha\,y = \dfrac{c}{a}$ avec $\alpha = \dfrac{b}{a}$.
  \item \textbf{Équation homogène} $y' + \alpha\,y = 0$ : ses solutions sont $y_h(t) = C\,e^{-\alpha t}$, $C$ constante réelle.
  \item \textbf{Solution particulière constante} (si $b \neq 0$) : $y_p = \dfrac{c}{b}$.
  \item \textbf{Solution générale} : $y(t) = y_h(t) + y_p = C\,e^{-\alpha t} + \dfrac{c}{b}$.
  \item \textbf{Condition initiale} $y(t_0) = y_0$ : remplacer $t$ par $t_0$, en déduire $C$, écrire la solution unique.
\end{enumerate}
\textbf{Constante de temps} (si $\alpha > 0$) : $\tau = \dfrac{1}{\alpha}$. À $t = \tau$, environ $63\,\%$ de l'évolution est réalisée.\\
\textbf{Exponentielle et logarithme} : $e^{0} = 1$ ; \ $\big(e^{kt}\big)' = k\,e^{kt}$ ; \ si $\alpha > 0$, $e^{-\alpha t} \to 0$ quand $t \to +\infty$ ; \ pour $A > 0$ : $e^{X} = A \iff X = \ln A$ ; \ $\ln\dfrac{1}{A} = -\ln A$.
\end{tcolorbox}

\exo{Réduire avant de résoudre}{C4 C5}{[S01] [C01] [C02]}{10 points}
% ============================================================
La fonction inconnue $y$ est une fonction de la variable $t$. Il n'y a pas de condition initiale dans cet exercice. Pour chaque équation :

\begin{enumerate}[label=\alph*), leftmargin=1.2cm, itemsep=1pt, topsep=2pt]
  \item la \textbf{réduire} : l'écrire sous la forme $a\,y' + b\,y = c$ et donner $a$, $b$ et $c$ \hfill \mbox{\small\color{uimmgray}(1 pt)}
  \item calculer $\alpha = \dfrac{b}{a}$, puis donner la \textbf{solution générale}. \hfill \mbox{\small\color{uimmgray}(1 pt)}
\end{enumerate}
Les résultats sont donnés sous forme exacte (fraction simplifiée ou décimal exact). Détailler les calculs de réduction.

\medskip
\begin{enumerate}
  \needspace{5.4cm}
  \item $5y' - 2y = 3y' + 4y$. \pts{2 pts}
  \rep{4.0}
  \needspace{5.4cm}
  \item $3\,(y' - 2y) = 2y' + 9$. \pts{2 pts}
  \rep{4.0}
  \needspace{5.4cm}
  \item $\dfrac{y'}{2} - \dfrac{y}{3} = 1$. \pts{2 pts}
  \rep{4.0}
  \needspace{5.4cm}
  \item $2y - y' = 4 + y$. \pts{2 pts}
  \rep{4.0}
  \needspace{5.4cm}
  \item $4 - 3y' = 2\,(y + 5) - y'$. \pts{2 pts}
  \rep{4.0}
\end{enumerate}

\exo{Mise sous tension d'un condensateur}{C4 C5 C6}{[S01] [C01] [C02]}{10 points}
% ============================================================
Un condensateur, chargé initialement sous $2$\,V, est relié à une source de tension continue à l'instant $t = 0$. La tension $u(t)$ à ses bornes (en volts, $t$ étant en secondes) vérifie, pour $t \geq 0$, l'équation
\[ (E) : \quad \dfrac{u'}{2} = 6 - u \qquad \text{avec} \qquad u(0) = 2 . \]

\begin{enumerate}
  \needspace{3.6cm}
  \item Montrer que $(E)$ équivaut à $u' + 2u = 12$. \pts{1 pt}
  \rep{2.4}
  \needspace{5.1cm}
  \item Donner la solution générale de $u' + 2u = 12$ (équation homogène, solution particulière constante, puis somme). \pts{2 pts}
  \rep{3.4}
  \needspace{4.6cm}
  \item Déterminer la constante $C$ grâce à $u(0) = 2$, puis montrer que $u(t) = 6 - 4\,e^{-2t}$. \pts{1,5 pt}
  \rep{3.0}
  \needspace{3.6cm}
  \item Déterminer la limite de $u(t)$ quand $t \to +\infty$, puis la constante de temps $\tau$. \pts{1 pt}
  \rep{2.0}
  \needspace{3.6cm}
  \item Calculer $u(\tau)$, arrondi au centième. \pts{1 pt}
  \rep{2.0}
  \needspace{5.6cm}
  \item Déterminer par le calcul l'instant $t_1$ tel que $u(t_1) = 5$\,V. Donner la valeur exacte, puis la valeur arrondie au centième de seconde. \pts{2 pts}
  \rep{4.0}
  \needspace{5.6cm}
  \item Pour une autre mise en service, avec la même équation $(E)$, on mesure $u(1) = 5{,}5$\,V. Déterminer la tension initiale $u(0)$, arrondie au centième de volt. \pts{1,5 pt}
  \rep{4.0}
\end{enumerate}
