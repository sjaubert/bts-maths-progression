\begin{tcolorbox}[colback=uimmlightblue!40, colframe=uimmblue, boxrule=0.8pt, arc=2pt, left=8pt, right=8pt, top=4pt, bottom=4pt, title={\bfseries Rappel : résoudre $a\,y' + b\,y = c$ ($a$, $b$, $c$ constantes, $a \neq 0$)}, coltitle=white, colbacktitle=uimmblue]
\small
\begin{enumerate}[leftmargin=*, itemsep=1pt, topsep=1pt]
  \item \textbf{Réduire l'équation} : tous les termes en $y'$ et en $y$ à gauche, la constante à droite. Lire $a$, $b$ et $c$.
  \item \textbf{Forme normalisée} : diviser par $a$, on obtient $y' + \alpha\,y = \dfrac{c}{a}$ avec $\alpha = \dfrac{b}{a}$.
  \item \textbf{Équation homogène} $y' + \alpha\,y = 0$ : ses solutions sont $y_h(t) = C\,e^{-\alpha t}$, $C$ constante réelle.
  \item \textbf{Solution particulière constante} (si $b \neq 0$) : $y_p = \dfrac{c}{b}$.
  \item \textbf{Solution générale} : $y(t) = y_h(t) + y_p = C\,e^{-\alpha t} + \dfrac{c}{b}$.
  \item \textbf{Condition initiale} $y(t_0) = y_0$ : remplacer $t$ par $t_0$, en déduire $C$, écrire la solution unique.
\end{enumerate}
\textbf{Constante de temps} (si $\alpha > 0$) : $\tau = \dfrac{1}{\alpha}$. À $t = \tau$, environ $63\,\%$ de l'évolution est réalisée.\\
\textbf{Exponentielle et logarithme} : $e^{0} = 1$ ; \ $\big(e^{kt}\big)' = k\,e^{kt}$ ; \ si $\alpha > 0$, $e^{-\alpha t} \to 0$ quand $t \to +\infty$ ; \ pour $A > 0$ : $e^{X} = A \iff X = \ln A$ ; \ $\ln\dfrac{1}{A} = -\ln A$.
\end{tcolorbox}

\exo{Résolution d'équations différentielles}{C4 C5}{[S01] [C01] [C02]}{11 points}
% ============================================================
La fonction inconnue $y$ est une fonction de la variable $t$. Pour chaque équation : la réduire, la résoudre, puis déterminer la solution qui vérifie la condition initiale.

\medskip
\textbf{\color{uimmblue}Partie A : équations homogènes}
\begin{enumerate}
  \needspace{5.1cm}
  \item $y' = y$ \quad et \quad $y(0) = 2$. \pts{1 pt}
  \rep{3.5}
  \needspace{5.6cm}
  \item $3y' + 2y = y' - 4y$ \quad et \quad $y(0) = 5$. \pts{1,5 pt}
  \rep{4.0}
  \needspace{5.1cm}
  \item $4y' = y$ \quad et \quad $y(0) = -1$. \pts{1 pt}
  \rep{3.5}
  \needspace{6.1cm}
  \item $\dfrac{y'}{2} + y = 0$ \quad et \quad $y(1) = 3$ \quad (la condition n'est pas donnée en $t = 0$). \pts{1,5 pt}
  \rep{4.5}
\end{enumerate}

\needspace{7cm}\medskip
\textbf{\color{uimmblue}Partie B : second membre constant}
\begin{enumerate}[resume]
  \needspace{5.6cm}
  \item $y' + 4y = 8$ \quad et \quad $y(0) = 0$. \pts{1 pt}
  \rep{4.0}
  \needspace{6.1cm}
  \item $2y' = y' - 3y + 9$ \quad et \quad $y(0) = 1$. \pts{1,5 pt}
  \rep{4.5}
  \needspace{6.1cm}
  \item $\dfrac{y'}{3} = \dfrac{y'}{2} + 1$ \quad et \quad $y(0) = 4$. \pts{1,5 pt}
  \rep{4.5}
  \needspace{6.6cm}
  \item $5\,(y - 2) = 2y'$ \quad et \quad $y(0) = 3$. \pts{2 pts}
  \rep{5.0}
\end{enumerate}

\exo{Essai de surcharge d'un moteur de convoyeur}{C3 C4 C5 C6}{[S01] [C01] [C02]}{9 points}
% ============================================================
Le service maintenance réalise un essai de surcharge sur le moteur asynchrone d'un convoyeur. On suit la température $\theta(t)$ de son bobinage (en $^\circ$C), $t$ étant le temps en minutes. Au départ, le moteur est froid : $\theta(0) = 20\,^\circ$C.

On admet le modèle suivant : la vitesse d'échauffement $\theta'(t)$ est \textbf{proportionnelle} à l'écart entre la température limite $\theta_L = 120\,^\circ$C et la température $\theta(t)$, avec le coefficient de proportionnalité $k = 0{,}025\ \text{min}^{-1}$.

Une sonde de protection, noyée dans le bobinage, arrête le moteur si sa température atteint $105\,^\circ$C (seuil retenu pour cet essai).

\begin{enumerate}
  \needspace{4.6cm}
  \item Traduire le modèle par une égalité entre $\theta'(t)$ et $\theta(t)$. Montrer que $\theta$ vérifie l'équation \[ (E) : \quad \theta' + 0{,}025\,\theta = 3 . \] \pts{1 pt}
  \rep{3.0}
  \needspace{3.1cm}
  \item Résoudre l'équation homogène associée à $(E)$. \pts{0,5 pt}
  \rep{1.5}
  \needspace{3.1cm}
  \item Déterminer une solution particulière constante de $(E)$. \pts{0,5 pt}
  \rep{1.5}
  \needspace{5.6cm}
  \item En déduire la solution générale de $(E)$, puis montrer que $\theta(t) = 120 - 100\,e^{-0{,}025\,t}$. \pts{1,5 pt}
  \rep{4.0}
  \needspace{4.6cm}
  \item Calculer $\theta'(t)$. Étudier son signe et en déduire le sens de variation de $\theta$ sur $[0\,;\,+\infty[$. Interpréter. \pts{1 pt}
  \rep{3.0}
  \needspace{3.6cm}
  \item Déterminer la limite de $\theta(t)$ quand $t$ tend vers $+\infty$. Interpréter ce résultat pour le moteur. \pts{1 pt}
  \rep{2.0}
  \needspace{4.6cm}
  \item On note $\tau = \dfrac{1}{0{,}025}$ la constante de temps. Calculer $\tau$, puis $\theta(\tau)$ arrondi au dixième. Quel pourcentage de l'échauffement total (de $20\,^\circ$C à la limite) est réalisé à cet instant ? \pts{1 pt}
  \rep{3.0}
  \needspace{5.6cm}
  \item Déterminer par le calcul l'instant $t_1$ où la sonde arrête le moteur. Arrondir à la minute. \pts{1,5 pt}
  \rep{4.0}
  \needspace{4.6cm}
  \item Le cycle de surcharge prévu dure $60$ minutes. Le moteur sera-t-il arrêté par la sonde avant la fin du cycle ? Justifier par un calcul. \pts{1 pt}
  \rep{3.0}
\end{enumerate}
