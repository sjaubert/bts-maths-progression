\section{Partie 1 -- Réception de détecteurs inductifs (8 points)}

\needspace{6cm}
\begin{correction}[-- Question 1 \pts{1,5 pt}]
\source{\LD{}, §2.1 et §2.2 Épreuve et schéma de Bernoulli p.~5~; §3.1 Définition et méthode «~Reconnaître une loi binomiale~» p.~7}
\begin{itemize}[leftmargin=12pt, itemsep=1pt, topsep=2pt]
  \item Épreuve de Bernoulli~: on teste un détecteur et on regarde s'il est défectueux. Deux issues. Succès~: «~le détecteur est défectueux~», de probabilité $p = 0{,}05$.
  \item L'épreuve est répétée $25$ fois, de façon identique et indépendante (prélèvement assimilé à un tirage avec remise).
  \item $X$ compte le nombre de succès.
\end{itemize}
Donc $X$ suit la loi binomiale $\mathcal{B}(25\,;0{,}05)$.
\bareme{Bernoulli, succès et $p$ : 0,5 \quad répétitions identiques et indépendantes : 0,5 \quad loi et paramètres : 0,5}
\piege{prendre $p = 0{,}95$ alors que $X$ compte les détecteurs \textbf{défectueux} ; oublier l'indépendance.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 2 \pts{1,5 pt}]
\source{\LD{}, §3.3 p.~8 (propriété «~Espérance et écart-type~»)}
\[
E(X) = np = 25\times 0{,}05 = 1{,}25,\quad \sigma(X) = \sqrt{np(1-p)} = \sqrt{25\times 0{,}05\times 0{,}95} = \sqrt{1{,}1875}\approx 1{,}0897.
\]
Interprétation~: sur un grand nombre de prélèvements de $25$ détecteurs, on trouve en moyenne $1{,}25$ détecteur défectueux par prélèvement.
\bareme{$E(X)$ : 0,5 \quad $\sigma(X)$ : 0,5 \quad interprétation : 0,5}
\piege{oublier la racine carrée (donner la variance $1{,}1875$) ; interpréter $1{,}25$ comme le nombre de défectueux de \textbf{ce} prélèvement.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3 \pts{1 pt}]
\source{\LD{}, §3.2 Coefficient binomial p.~7~; §3.3 Calcul des probabilités p.~7 et exemple «~Calcul détaillé~» p.~8}
L'évènement est $\{X = 2\}$~: $2$ détecteurs défectueux et $23$ corrects. Le coefficient $\binom{25}{2} = \dfrac{25\times 24}{2} = 300$ compte les placements des $2$ défectueux dans le prélèvement.
\[
P(X = 2) = \binom{25}{2}\times 0{,}05^{2}\times 0{,}95^{23} = 300\times 0{,}0025\times 0{,}95^{23}\approx 0{,}75\times 0{,}3074\approx 0{,}2305.
\]
\bareme{coefficient $300$ : 0,25 \quad formule complète : 0,5 \quad résultat : 0,25}
\piege{écrire $0{,}95^{25}$ au lieu de $0{,}95^{23}$ ; oublier le coefficient $300$ ; calculer $25\times 24 = 600$ sans diviser par $2$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 4.a \pts{0,75 pt}]
\source{\LD{}, §3.3 Calcul des probabilités p.~7 (terme général)}
«~Aucun détecteur défectueux~» $\iff X = 0$.
\[
P(X = 0) = 0{,}95^{25}\approx 0{,}2774.
\]
\bareme{traduction : 0,25 \quad résultat : 0,5}
\piege{écrire $0{,}05^{25}$ (probabilité que \textbf{tous} soient défectueux).}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 4.b \pts{0,75 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~» et exemple $P(X\geq 1) = 1 - P(X = 0)$)}
«~Au moins un défectueux~» $\iff X\geq 1$. Son contraire est $X = 0$.
\[
P(X\geq 1) = 1 - P(X = 0)\approx 1 - 0{,}2774 = 0{,}7226.
\]
\bareme{traduction et contraire : 0,25 \quad résultat : 0,5}
\piege{calculer $P(X = 1)\approx 0{,}3650$.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4.c \pts{1 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~»)}
«~Au moins $23$ corrects~» $\iff$ «~$23$, $24$ ou $25$ corrects~» $\iff$ «~$2$, $1$ ou $0$ défectueux~» $\iff X\leq 2$.
\[
P(X\leq 2) = \texttt{binomFRép}(25,\,0{,}05,\,2)\approx 0{,}8729.
\]
\bareme{traduction $X\leq 2$ : 0,5 \quad résultat : 0,5}
\piege{écrire $X\geq 23$ (probabilité quasi nulle, qui devrait alerter) ; écrire $X < 2$ ($\approx 0{,}6424$).}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 4.d \pts{1 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~»)}
«~Au plus $3$ défectueux~» $\iff X\leq 3$.
\[
P(X\leq 3) = \texttt{binomFRép}(25,\,0{,}05,\,3)\approx 0{,}9659.
\]
\bareme{traduction : 0,5 \quad résultat : 0,5}
\piege{traduire par $X < 3$ ($\approx 0{,}8729$) ; utiliser \texttt{binomFdp} au lieu de \texttt{binomFRép} ($\approx 0{,}0930$).}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 5 \pts{0,5 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~» : $P(X\geq k) = 1 - P(X\leq k-1)$)}
«~Plus de $3$~» signifie «~strictement plus de $3$~»~: $X\geq 4$. C'est le contraire de la question 4.d.
\[
P(X\geq 4) = 1 - P(X\leq 3)\approx 1 - 0{,}9659 = 0{,}0341.
\]
Un arrivage conforme à l'annonce du fournisseur est refusé dans environ $3{,}4\,\%$ des cas.
\bareme{traduction $X\geq 4$ et contraire : 0,25 \quad résultat : 0,25}
\piege{traduire par $X\geq 3$ ($\approx 0{,}1271$) ; écrire $1 - P(X\leq 4)$ ($\approx 0{,}0072$).}
\end{correction}

\section{Partie 2 -- Loi de Poisson (8 points)}
\subsection*{A. Pertes de communication sur le réseau de terrain (5 points)}

\needspace{6cm}
\begin{correction}[-- Question 1 \pts{1 pt}]
\source{\LD{}, §4.1 Définition p.~9}
Soit $Y$ le nombre de pertes de communication enregistrées par l'automate au cours d'une semaine de production. $Y$ suit une loi de Poisson $\mathcal{P}(\lambda)$.\\
Pour une loi de Poisson, $E(Y) = \lambda$. Or le nombre moyen de pertes est $2{,}5$ par semaine. Donc $\lambda = 2{,}5$ et $Y$ suit $\mathcal{P}(2{,}5)$.
\bareme{variable définie en phrase (avec la durée : une semaine) : 0,5 \quad $\lambda = E(Y) = 2{,}5$ justifié : 0,5}
\piege{variable non définie ou sans unité de temps ; paramètre donné sans justification.}
\end{correction}

\needspace{4cm}
\begin{correction}[-- Question 2 \pts{0,5 pt}]
\source{\LD{}, §4.1 p.~9 (propriété «~Espérance et écart-type~»)}
$\sigma(Y) = \sqrt{\lambda} = \sqrt{2{,}5}\approx 1{,}5811$.
\bareme{résultat : 0,5}
\piege{répondre $\sigma = 2{,}5$ (confusion avec la variance).}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3 \pts{1 pt}]
\source{\LD{}, §4.1 p.~9 (exemple «~Calcul détaillé~»)}
\[
P(Y = 0) = e^{-2{,}5}\,\frac{2{,}5^{0}}{0!} = e^{-2{,}5}\approx 0{,}0821,\qquad
P(Y = 3) = e^{-2{,}5}\,\frac{2{,}5^{3}}{3!} = \frac{15{,}625}{6}\,e^{-2{,}5}\approx 0{,}2138.
\]
\bareme{$P(Y = 0)$ avec formule : 0,5 \quad $P(Y = 3)$ avec formule : 0,5}
\piege{écrire $2{,}5^{0} = 0$ ; oublier la factorielle ($15{,}625\,e^{-2{,}5}\approx 1{,}2826$, une probabilité supérieure à $1$ qui devrait alerter) ; écrire $3! = 3$.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4 \pts{1 pt}]
\source{\LD{}, §4.1 p.~9 (encadré «~À la calculatrice~»)~; règle $P(X\geq k) = 1 - P(X\leq k-1)$ : §3.3 p.~8}
«~Plus de $4$ pertes~» $\iff Y\geq 5$. Son contraire est $Y\leq 4$.
\[
P(Y\geq 5) = 1 - P(Y\leq 4) = 1 - \texttt{poissonFRép}(2{,}5,\,4)\approx 1 - 0{,}8912 = 0{,}1088.
\]
\bareme{traduction et contraire : 0,5 \quad résultat : 0,5}
\piege{traduire par $Y\geq 4$ ($\approx 0{,}2424$) ; écrire $1 - P(Y\leq 5)$ ($\approx 0{,}0420$).}
\end{correction}

\needspace{7cm}
\begin{correction}[-- Question 5 \pts{1,5 pt}]
\source{\LD{}, §4.1 p.~9 (encadré «~À la calculatrice~»)}
On cherche le plus petit entier $k$ tel que $P(Y\leq k)\geq 0{,}95$. On calcule les probabilités cumulées~:
\[
P(Y\leq 4)\approx 0{,}8912 < 0{,}95 \qquad\text{et}\qquad P(Y\leq 5)\approx 0{,}9580\geq 0{,}95.
\]
Donc $k = 5$~: le diagnostic complet n'est déclenché que les semaines à plus de $5$ pertes, soit environ $4\,\%$ des semaines.
\bareme{traduction $P(Y\leq k)\geq 0{,}95$ : 0,5 \quad les deux probabilités encadrantes : 0,5 \quad conclusion $k = 5$ : 0,5}
\piege{donner une seule probabilité (sans montrer que $k = 4$ ne convient pas) ; chercher $P(Y = k)\geq 0{,}95$ ; répondre $k = 4$.}
\end{correction}

\subsection*{B. Approximation d'une loi binomiale par une loi de Poisson (3 points)}

\needspace{5cm}
\begin{correction}[-- Question 6 \pts{0,5 pt}]
\source{\LD{}, §3.1 Définition p.~7}
Chaque carte est défectueuse ou non, avec la probabilité $0{,}0075$, de façon indépendante, sur $400$ cartes. $N$ compte les cartes défectueuses~: $N$ suit la loi binomiale $\mathcal{B}(400\,;0{,}0075)$.
\bareme{loi et paramètres : 0,5}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 7 \pts{1 pt}]
\source{\LD{}, §5 Approximation d'une loi binomiale par une loi de Poisson, propriété «~Approximation de Poisson~» p.~11}
\begin{itemize}[leftmargin=12pt, itemsep=1pt, topsep=2pt]
  \item $n = 400\geq 30$~;
  \item $p = 0{,}0075\leq 0{,}1$~;
  \item $np = 400\times 0{,}0075 = 3\leq 10$.
\end{itemize}
Les trois conditions sont réunies. On peut approcher la loi de $N$ par la loi de Poisson $\mathcal{P}(3)$, avec $\lambda = np = 3$.
\bareme{chaque condition vérifiée avec sa valeur : 3 $\times$ 0,25 \quad $\lambda = 3$ : 0,25}
\piege{vérifier une seule condition ; écrire «~$n$ grand, $p$ petit~» sans valeurs ; prendre $\lambda = 0{,}0075$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 8 \pts{1 pt}]
\source{\LD{}, §4.1 p.~9 (encadré «~À la calculatrice~»)~; §5 p.~11}
Soit $N'$ de loi $\mathcal{P}(3)$. «~Au plus $2$ cartes défectueuses~» $\iff N'\leq 2$.
\[
P(N'\leq 2) = \texttt{poissonFRép}(3,\,2)\approx 0{,}4232.
\]
\bareme{traduction : 0,5 \quad résultat : 0,5}
\piege{calculer $P(N' = 2)\approx 0{,}2240$ ; calculer $P(N' < 2)\approx 0{,}1991$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 9 \pts{0,5 pt}]
\source{\LD{}, §5 p.~11 (exemple «~Comparaison chiffrée~»)}
Avec la loi exacte~: $P(N\leq 2) = \texttt{binomFRép}(400,\,0{,}0075,\,2)\approx 0{,}4223$.\\
L'écart est d'environ $0{,}0009$, soit moins de $0{,}1$ point de pourcentage. L'approximation est très bonne.
\bareme{valeur exacte : 0,25 \quad comparaison et conclusion : 0,25}
\piege{conclure sans chiffrer l'écart.}
\end{correction}

\bareme{intervalle : 0,5 \quad argument de symétrie et cohérence : 0,5}

\end{correction}

\needspace{5cm}

\bareme{produit $1\,000\times P(T > 12{,}8)$ : 0,5 \quad résultat arrondi et interprété : 0,5}
\piege{répondre $0{,}0228$ ; utiliser $P(T\leq 12{,}8)$ ($\approx 977$ cycles).}
\end{correction}
