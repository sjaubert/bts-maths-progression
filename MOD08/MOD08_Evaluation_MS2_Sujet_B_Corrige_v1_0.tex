\section{Exercice 1 -- Joints d'étanchéité (8 points)}

\needspace{6cm}
\begin{correction}[-- Question 1 \pts{1 pt}]
\source{\VM{} (arbre pondéré, probabilités conditionnelles)}
On note $P_1$ l'évènement «~le joint a été moulé sur la presse P1~», $\overline{P_1}$ son contraire («~le joint a été moulé sur la presse P2~») et $N$ l'évènement «~le joint est non conforme~».
\[
P(P_1) = 0{,}7,\qquad P(\overline{P_1}) = 0{,}3,\qquad P_{P_1}(N) = 0{,}03,\qquad P_{\overline{P_1}}(N) = 0{,}08.
\]
\bareme{évènements définis en phrase : 0,5 \quad probabilités conditionnelles correctement notées : 0,5}
\piege{écrire $P(P_1\cap N) = 0{,}03$ au lieu de $P_{P_1}(N) = 0{,}03$ ; oublier de définir les évènements.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 2 \pts{1,5 pt}]
\source{\VM{} (formule des probabilités totales)}
$P_1$ et $\overline{P_1}$ forment une partition de l'univers. D'après la formule des probabilités totales :
\[
P(N) = P(P_1)\,P_{P_1}(N) + P(\overline{P_1})\,P_{\overline{P_1}}(N) = 0{,}7\times 0{,}03 + 0{,}3\times 0{,}08 = 0{,}021 + 0{,}024 = 0{,}045.
\]
\bareme{partition ou formule citée : 0,5 \quad calcul : 1}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3 \pts{1,5 pt}]
\source{\VM{} (probabilité conditionnelle «~inversée~»)}
On cherche $P_N(\overline{P_1})$ :
\[
P_N(\overline{P_1}) = \frac{P(\overline{P_1}\cap N)}{P(N)} = \frac{0{,}024}{0{,}045} \approx 0{,}5333.
\]
Un joint non conforme a été moulé sur la presse P2 avec une probabilité d'environ $0{,}5333$ : P2 fabrique $30\,\%$ des joints mais plus de la moitié des non conformes.
\bareme{formule : 0,5 \quad numérateur $P(\overline{P_1}\cap N)$ : 0,5 \quad résultat et phrase : 0,5}
\piege{répondre $0{,}08$, c'est-à-dire confondre $P_N(\overline{P_1})$ et $P_{\overline{P_1}}(N)$.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4 \pts{1,5 pt}]
\source{\LD{}, §2.2 Schéma de Bernoulli p.~5 et §3.1 Définition p.~7}
\begin{itemize}[leftmargin=12pt, itemsep=1pt, topsep=2pt]
  \item Épreuve de Bernoulli : on prélève un joint et on regarde s'il est conforme. Deux issues. Succès : «~le joint est conforme~», de probabilité $p = 1 - 0{,}045 = 0{,}955$.
  \item L'épreuve est répétée $30$ fois, de façon identique et indépendante (tirage assimilé à un tirage avec remise).
  \item $X$ compte le nombre de succès.
\end{itemize}
Donc $X$ suit la loi binomiale $\mathcal{B}(30\,;0{,}955)$.
\bareme{Bernoulli, succès et $p$ : 0,5 \quad répétitions identiques et indépendantes : 0,5 \quad loi et paramètres : 0,5}
\piege{prendre $p = 0{,}045$ alors que $X$ compte les \textbf{conformes} ; oublier l'indépendance.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 5 \pts{1,5 pt}]
\source{\LD{}, §3.3 Calcul des probabilités p.~7 (encadré «~À la calculatrice~»)}
«~Au plus $3$ non conformes~» $\iff$ «~au moins $30 - 3 = 27$ conformes~» $\iff X \geq 27$.
\[
P(X\geq 27) = 1 - P(X\leq 26) = 1 - \texttt{binomFRép}(30,\,0{,}955,\,26) \approx 1 - 0{,}0442 = 0{,}9558.
\]
Variante acceptée : $Y = 30 - X$ suit $\mathcal{B}(30\,;0{,}045)$ et $P(Y\leq 3) \approx 0{,}9558$, à condition que $Y$ soit défini.
\bareme{traduction $X\geq 27$ : 0,75 \quad passage au complémentaire et résultat : 0,75}
\piege{calculer $P(X\leq 3)$ (résultat quasi nul, qui devrait alerter) ; écrire $1 - P(X\leq 27)$ ; traduire par $X \geq 26$.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 6 \pts{1 pt}]
\source{\LD{}, §3.3 p.~8 (propriété «~Espérance et écart-type~»)}
Le nombre de non conformes est $Y = 30 - X$, donc $E(Y) = 30 - E(X) = 30 - 30\times 0{,}955 = 30 - 28{,}65 = 1{,}35$.\\
On peut aussi dire que $Y$ suit $\mathcal{B}(30\,;0{,}045)$, donc $E(Y) = 30\times 0{,}045 = 1{,}35$.\\
En moyenne, un prélèvement de $30$ joints contient $1{,}35$ joint non conforme.
\bareme{justification ($E = np$ appliquée au bon $p$) : 0,5 \quad valeur et phrase : 0,5}
\piege{répondre $E(X) = 28{,}65$, qui est le nombre moyen de \textbf{conformes}.}
\end{correction}

\section{Exercice 2 -- Pannes de variateurs (5 points)}

\needspace{6cm}
\begin{correction}[-- Question 1 \pts{1 pt}]
\source{\LD{}, §5.1 Principe p.~11}
$n = 200 \geq 30$, $p = 0{,}01 \leq 0{,}1$ et $np = 2 \leq 10$. On peut approcher la loi de $N$ par la loi de Poisson $\mathcal{P}(\lambda)$ avec $\lambda = np = 2$.
\bareme{les trois conditions : 0,5 \quad $\lambda = np$ : 0,5}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 2 \pts{1,5 pt}]
\source{\LD{}, §4.1 Définition p.~9 (encadré «~À la calculatrice~»)}
$P(N = 0) = e^{-2} \approx 0{,}1353$.
\qquad
$P(N\leq 2) = \texttt{poissonFRép}(2,\,2) \approx 0{,}6767$.\\
Calcul à la main accepté : $P(N\leq 2) = e^{-2}\left(1 + 2 + \dfrac{2^2}{2}\right) \approx 0{,}6767$.
\bareme{$P(N=0)$ : 0,5 \quad $P(N\leq 2)$ : 1}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3 \pts{1,5 pt}]
\source{\VM{} (définition de $P_B(A)$) et \LD{}, §4.1 p.~9}
On cherche $P_{\{N\geq 1\}}(N\leq 2)$. Or $\{N\leq 2\}\cap\{N\geq 1\} = \{1\leq N\leq 2\}$.
\[
P_{\{N\geq 1\}}(N\leq 2) = \frac{P(1\leq N\leq 2)}{P(N\geq 1)} = \frac{P(N\leq 2) - P(N=0)}{1 - P(N=0)} \approx \frac{0{,}6767 - 0{,}1353}{1 - 0{,}1353} \approx 0{,}6261.
\]
Sachant qu'il y a eu au moins une panne, la probabilité qu'il y en ait eu au plus deux vaut environ $0{,}6261$.
\bareme{formule de la probabilité conditionnelle : 0,5 \quad intersection $\{1\leq N\leq 2\}$ : 0,5 \quad résultat : 0,5}
\piege{répondre $P(N\leq 2)$ sans conditionner ; faire le produit $P(N\leq 2)\times P(N\geq 1)$ ; garder $P(N=0)$ au numérateur.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4 \pts{1 pt}]
\source{\LD{}, §4.1 p.~9}
La réserve suffit si $N \leq k$. On cherche le plus petit $k$ tel que $P(N\leq k)\geq 0{,}95$.\\
$P(N\leq 4) \approx 0{,}9473 < 0{,}95$ \quad et \quad $P(N\leq 5) \approx 0{,}9834 \geq 0{,}95$. Donc $k = 5$.
\bareme{les deux valeurs encadrantes : 0,5 \quad conclusion : 0,5}
\piege{s'arrêter à la première valeur calculée sans montrer que la précédente ne convient pas.}
\end{correction}

\noindent{\footnotesize\color{uimmgray}\textit{Contrôle : avec la loi binomiale exacte $\mathcal{B}(200\,;0{,}01)$, on trouve $P(N\leq 4)\approx 0{,}9483$ et $P(N\leq 5)\approx 0{,}984$ : la valeur de $k$ est la même.}}

\section{Exercice 3 -- Longueur d'entretoises (7 points)}

\needspace{6cm}
\begin{correction}[-- Question 1 \pts{1,5 pt}]
\source{\LN{}, §3.2 Calcul d'une probabilité p.~6}
$L$ suit $\mathcal{N}(120\,;0{,}25)$.
\[
P(119{,}4 \leq L \leq 120{,}6) = \texttt{normalFRép}(119{,}4,\,120{,}6,\,120,\,0{,}25) \approx 0{,}9836.
\]
Ces bornes valent $\mu \pm 2{,}4\sigma$.
\bareme{probabilité écrite avec $L$ : 0,5 \quad commande et paramètres : 0,5 \quad résultat : 0,5}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 2 \pts{1 pt}]
\source{\LN{}, §3.2 p.~6 et §3.3 Standardisation p.~7}
$P(L < 119{,}8) = \texttt{normalFRép}(-10^{99},\,119{,}8,\,120,\,0{,}25) \approx 0{,}2119$.\\ Par standardisation : $\dfrac{119{,}8 - 120}{0{,}25} = -0{,}8$ et $P(L < 119{,}8) = P(Z\leq -0{,}8) = 1 - P(Z\leq 0{,}8)\approx 0{,}2119$.
\bareme{écriture correcte (sens de l'inégalité) : 0,5 \quad résultat : 0,5}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3 \pts{2 pts}]
\source{\VM{} (définition de $P_B(A)$) et \LN{}, §3.2 p.~6}
Soit $R$ l'évènement «~l'entretoise est rebutée~» et $\mathcal{A}$ l'évènement «~l'entretoise est en zone d'alerte~». On cherche $P_{\mathcal{A}}(R)$.\\
$R = \{L < 119{,}4\} \cup \{L > 120{,}6\}$ et $\mathcal{A} = \{L < 119{,}8\}$. Donc $R\cap \mathcal{A} = \{L < 119{,}4\}$.
\[
P_{\mathcal{A}}(R) = \frac{P(R\cap \mathcal{A})}{P(\mathcal{A})} = \frac{P(L < 119{,}4)}{P(L < 119{,}8)} \approx \frac{0{,}0082}{0{,}2119} \approx 0{,}0387.
\]
Moins de $4\,\%$ des entretoises en zone d'alerte sont rebutées ($P_{\mathcal{A}}(R)\approx 0{,}0387$).
\bareme{évènements et formule : 0,5 \quad intersection $R\cap \mathcal{A} = \{L < 119{,}4\}$ : 0,75 \quad calcul et phrase : 0,75}
\piege{répondre $P(R) \approx 0{,}0164$ sans conditionner ; oublier que la zone d'alerte ne contient qu'une des deux queues de rebut.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4 \pts{2,5 pts}]
\source{\LN{}, §3.3 Standardisation p.~7 et §4.2 Retrouver un paramètre p.~8}
On pose $Z = \dfrac{L - 120}{\sigma}$, qui suit $\mathcal{N}(0\,;1)$. La tolérance vaut $\pm 0{,}6$ autour de la moyenne.
\begin{align*}
P(119{,}4\leq L\leq 120{,}6)\geq 0{,}99
&\iff P\!\left(-\frac{0{,}6}{\sigma}\leq Z\leq \frac{0{,}6}{\sigma}\right)\geq 0{,}99\\
&\iff 2\,P\!\left(Z\leq \frac{0{,}6}{\sigma}\right) - 1\geq 0{,}99\\
&\iff P\!\left(Z\leq \frac{0{,}6}{\sigma}\right)\geq 0{,}995.
\end{align*}
$\texttt{FracNormale}(0{,}995)\approx 2{,}5758$, donc $\dfrac{0{,}6}{\sigma}\geq 2{,}5758$, soit $\sigma\leq \dfrac{0{,}6}{2{,}5758}\approx 0{,}2329$.\\
L'écart-type maximal est $\sigma\approx 0{,}2329$\,mm (arrondi \textbf{par défaut}, pour respecter l'exigence).
\bareme{standardisation : 0,75 \quad symétrie, passage à $0{,}995$ : 0,75 \quad $z = 2{,}5758$ : 0,5 \quad $\sigma$ et arrondi par défaut : 0,5}
\piege{utiliser $\texttt{FracNormale}(0{,}99)\approx 2{,}3263$ (oubli de la symétrie), ce qui donne $\sigma\approx 0{,}2579$ ; arrondir $\sigma$ par excès.}
\end{correction}
