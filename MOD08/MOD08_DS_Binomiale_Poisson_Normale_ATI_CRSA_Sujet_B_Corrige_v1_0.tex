\section{Partie 1 -- Remise en état de vérins pneumatiques (8 points)}

\needspace{6cm}
\begin{correction}[-- Question 1 \pts{1,5 pt}]
\source{\LD{}, §2.1 et §2.2 Épreuve et schéma de Bernoulli p.~5~; §3.1 Définition et méthode «~Reconnaître une loi binomiale~» p.~7}
\begin{itemize}[leftmargin=12pt, itemsep=1pt, topsep=2pt]
  \item Épreuve de Bernoulli~: on teste l'étanchéité d'un vérin. Deux issues. Succès~: «~le vérin fuit~», de probabilité $p = 0{,}08$.
  \item L'épreuve est répétée $20$ fois, de façon identique et indépendante (prélèvement assimilé à un tirage avec remise).
  \item $X$ compte le nombre de succès.
\end{itemize}
Donc $X$ suit la loi binomiale $\mathcal{B}(20\,;0{,}08)$.
\bareme{Bernoulli, succès et $p$ : 0,5 \quad répétitions identiques et indépendantes : 0,5 \quad loi et paramètres : 0,5}
\piege{prendre $p = 0{,}92$ alors que $X$ compte les vérins \textbf{qui fuient} ; oublier l'indépendance.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 2 \pts{1,5 pt}]
\source{\LD{}, §3.3 p.~8 (propriété «~Espérance et écart-type~»)}
\[
E(X) = np = 20\times 0{,}08 = 1{,}6,\quad \sigma(X) = \sqrt{np(1-p)} = \sqrt{20\times 0{,}08\times 0{,}92} = \sqrt{1{,}472}\approx 1{,}2133.
\]
Interprétation~: sur un grand nombre de prélèvements de $20$ vérins, on trouve en moyenne $1{,}6$ vérin qui fuit par prélèvement.
\bareme{$E(X)$ : 0,5 \quad $\sigma(X)$ : 0,5 \quad interprétation : 0,5}
\piege{oublier la racine carrée (donner la variance $1{,}472$) ; interpréter $1{,}6$ comme le nombre de vérins qui fuient dans \textbf{ce} prélèvement.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3 \pts{1 pt}]
\source{\LD{}, §3.2 Coefficient binomial p.~7~; §3.3 Calcul des probabilités p.~7 et exemple «~Calcul détaillé~» p.~8}
L'évènement est $\{X = 2\}$~: $2$ vérins qui fuient et $18$ étanches. Le coefficient $\binom{20}{2} = \dfrac{20\times 19}{2} = 190$ compte les placements des $2$ vérins qui fuient dans le prélèvement.
\[
P(X = 2) = \binom{20}{2}\times 0{,}08^{2}\times 0{,}92^{18} = 190\times 0{,}0064\times 0{,}92^{18}\approx 1{,}216\times 0{,}2229\approx 0{,}2711.
\]
\bareme{coefficient $190$ : 0,25 \quad formule complète : 0,5 \quad résultat : 0,25}
\piege{écrire $0{,}92^{20}$ au lieu de $0{,}92^{18}$ ; oublier le coefficient $190$ ; calculer $20\times 19 = 380$ sans diviser par $2$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 4.a \pts{0,75 pt}]
\source{\LD{}, §3.3 Calcul des probabilités p.~7 (terme général)}
«~Aucun vérin ne fuit~» $\iff X = 0$.
\[
P(X = 0) = 0{,}92^{20}\approx 0{,}1887.
\]
\bareme{traduction : 0,25 \quad résultat : 0,5}
\piege{écrire $0{,}08^{20}$ (probabilité que \textbf{tous} fuient).}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 4.b \pts{0,75 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~») et §3.4 p.~8 (application «~Contrôle qualité d'un lot~» : $P(X\geq 1) = 1 - P(X = 0)$)}
«~Au moins un vérin qui fuit~» $\iff X\geq 1$. Son contraire est $X = 0$.
\[
P(X\geq 1) = 1 - P(X = 0)\approx 1 - 0{,}1887 = 0{,}8113.
\]
\bareme{traduction et contraire : 0,25 \quad résultat : 0,5}
\piege{calculer $P(X = 1)\approx 0{,}3282$.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4.c \pts{1 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~»)}
«~Au moins $18$ étanches~» $\iff$ «~$18$, $19$ ou $20$ étanches~» $\iff$ «~$2$, $1$ ou $0$ qui fuient~» $\iff X\leq 2$.
\[
P(X\leq 2) = \texttt{binomFRép}(20,\,0{,}08,\,2)\approx 0{,}7879.
\]
\bareme{traduction $X\leq 2$ : 0,5 \quad résultat : 0,5}
\piege{écrire $X\geq 18$ (probabilité quasi nulle, qui devrait alerter) ; écrire $X < 2$ ($\approx 0{,}5169$).}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 4.d \pts{1 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~»)}
«~Au plus $3$ vérins qui fuient~» $\iff X\leq 3$.
\[
P(X\leq 3) = \texttt{binomFRép}(20,\,0{,}08,\,3)\approx 0{,}9294.
\]
\bareme{traduction : 0,5 \quad résultat : 0,5}
\piege{traduire par $X < 3$ ($\approx 0{,}7879$) ; utiliser \texttt{binomFdp} au lieu de \texttt{binomFRép} ($\approx 0{,}1414$).}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 5 \pts{0,5 pt}]
\source{\LD{}, §3.3 p.~8 (encadré «~À la calculatrice~» : $P(X\geq k) = 1 - P(X\leq k-1)$)}
«~Plus de $3$~» signifie «~strictement plus de $3$~»~: $X\geq 4$. C'est le contraire de la question 4.d.
\[
P(X\geq 4) = 1 - P(X\leq 3)\approx 1 - 0{,}9294 = 0{,}0706.
\]
Alors que la proportion de fuites reste de $8\,\%$, une analyse des causes est lancée dans environ $7\,\%$ des prélèvements.
\bareme{traduction $X\geq 4$ et contraire : 0,25 \quad résultat : 0,25}
\piege{traduire par $X\geq 3$ ($\approx 0{,}2121$) ; écrire $1 - P(X\leq 4)$ ($\approx 0{,}0183$).}
\end{correction}

\section{Partie 2 -- Loi de Poisson (8 points)}
\subsection*{A. Appels de l'astreinte maintenance (5 points)}

\needspace{6cm}
\begin{correction}[-- Question 1 \pts{1 pt}]
\source{\LD{}, §4.1 Définition p.~9}
Soit $Y$ le nombre d'appels de dépannage reçus par le technicien d'astreinte au cours d'une nuit. $Y$ suit une loi de Poisson $\mathcal{P}(\lambda)$.\\
Pour une loi de Poisson, $E(Y) = \lambda$. Or le nombre moyen d'appels est $1{,}8$ par nuit. Donc $\lambda = 1{,}8$ et $Y$ suit $\mathcal{P}(1{,}8)$.
\bareme{variable définie en phrase (avec la durée : une nuit) : 0,5 \quad $\lambda = E(Y) = 1{,}8$ justifié : 0,5}
\piege{variable non définie ou sans unité de temps ; paramètre donné sans justification.}
\end{correction}

\needspace{4cm}
\begin{correction}[-- Question 2 \pts{0,5 pt}]
\source{\LD{}, §4.1 p.~9 (propriété «~Espérance et écart-type~»)}
$\sigma(Y) = \sqrt{\lambda} = \sqrt{1{,}8}\approx 1{,}3416$.
\bareme{résultat : 0,5}
\piege{répondre $\sigma = 1{,}8$ (confusion avec la variance).}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3 \pts{1 pt}]
\source{\LD{}, §4.1 p.~9 (exemple «~Calcul détaillé~»)}
\[
P(Y = 0) = e^{-1{,}8}\,\frac{1{,}8^{0}}{0!} = e^{-1{,}8}\approx 0{,}1653,\qquad
P(Y = 2) = e^{-1{,}8}\,\frac{1{,}8^{2}}{2!} = 1{,}62\,e^{-1{,}8}\approx 0{,}2678.
\]
\bareme{$P(Y = 0)$ avec formule : 0,5 \quad $P(Y = 2)$ avec formule : 0,5}
\piege{écrire $1{,}8^{0} = 0$ ; oublier la factorielle ($3{,}24\,e^{-1{,}8}\approx 0{,}5356$) ; écrire $1{,}8^{2} = 3{,}6$.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4 \pts{1 pt}]
\source{\LD{}, §4.1 p.~9 (encadré «~À la calculatrice~»)~; règle $P(X\geq k) = 1 - P(X\leq k-1)$ : §3.3 p.~8}
«~Plus de $3$ appels~» $\iff Y\geq 4$. Son contraire est $Y\leq 3$.
\[
P(Y\geq 4) = 1 - P(Y\leq 3) = 1 - \texttt{poissonFRép}(1{,}8,\,3)\approx 1 - 0{,}8913 = 0{,}1087.
\]
\bareme{traduction et contraire : 0,5 \quad résultat : 0,5}
\piege{traduire par $Y\geq 3$ ($\approx 0{,}2694$) ; écrire $1 - P(Y\leq 4)$ ($\approx 0{,}0364$).}
\end{correction}

\needspace{7cm}
\begin{correction}[-- Question 5 \pts{1,5 pt}]
\source{\LD{}, §4.1 p.~9 (encadré «~À la calculatrice~»)}
On cherche le plus petit entier $k$ tel que $P(Y\leq k)\geq 0{,}95$. On calcule les probabilités cumulées~:
\[
P(Y\leq 3)\approx 0{,}8913 < 0{,}95 \qquad\text{et}\qquad P(Y\leq 4)\approx 0{,}9636\geq 0{,}95.
\]
Donc $k = 4$~: le renfort n'est rappelé que les nuits à plus de $4$ appels, soit environ $4\,\%$ des nuits.
\bareme{traduction $P(Y\leq k)\geq 0{,}95$ : 0,5 \quad les deux probabilités encadrantes : 0,5 \quad conclusion $k = 4$ : 0,5}
\piege{donner une seule probabilité (sans montrer que $k = 3$ ne convient pas) ; chercher $P(Y = k)\geq 0{,}95$ ; répondre $k = 3$.}
\end{correction}

\subsection*{B. Approximation d'une loi binomiale par une loi de Poisson (3 points)}

\needspace{5cm}
\begin{correction}[-- Question 6 \pts{0,5 pt}]
\source{\LD{}, §3.1 Définition p.~7}
Chaque câble est défectueux ou non, avec la probabilité $0{,}008$, de façon indépendante, sur $300$ câbles. $N$ compte les câbles défectueux~: $N$ suit la loi binomiale $\mathcal{B}(300\,;0{,}008)$.
\bareme{loi et paramètres : 0,5}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 7 \pts{1 pt}]
\source{\LD{}, §5.1 Principe p.~11 (propriété «~Approximation de Poisson~»)}
\begin{itemize}[leftmargin=12pt, itemsep=1pt, topsep=2pt]
  \item $n = 300\geq 30$~;
  \item $p = 0{,}008\leq 0{,}1$~;
  \item $np = 300\times 0{,}008 = 2{,}4\leq 10$.
\end{itemize}
Les trois conditions sont réunies. On peut approcher la loi de $N$ par la loi de Poisson $\mathcal{P}(2{,}4)$, avec $\lambda = np = 2{,}4$.
\bareme{chaque condition vérifiée avec sa valeur : 3 $\times$ 0,25 \quad $\lambda = 2{,}4$ : 0,25}
\piege{vérifier une seule condition ; écrire «~$n$ grand, $p$ petit~» sans valeurs ; prendre $\lambda = 0{,}008$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 8 \pts{1 pt}]
\source{\LD{}, §4.1 p.~9 (encadré «~À la calculatrice~»)~; §5.1 p.~11}
Soit $N'$ de loi $\mathcal{P}(2{,}4)$. «~Au plus $1$ câble défectueux~» $\iff N'\leq 1$.
\[
P(N'\leq 1) = \texttt{poissonFRép}(2{,}4,\,1)\approx 0{,}3084.
\]
\bareme{traduction : 0,5 \quad résultat : 0,5}
\piege{calculer $P(N' = 1)\approx 0{,}2177$ ; calculer $P(N' < 1)\approx 0{,}0907$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 9 \pts{0,5 pt}]
\source{\LD{}, §5.1 p.~11 (exemple «~Comparaison chiffrée~»)}
Avec la loi exacte~: $P(N\leq 1) = \texttt{binomFRép}(300,\,0{,}008,\,1)\approx 0{,}3072$.\\
L'écart est d'environ $0{,}0012$, soit environ $0{,}1$ point de pourcentage. L'approximation est très bonne.
\bareme{valeur exacte : 0,25 \quad comparaison et conclusion : 0,25}

\end{correction}
