\section{Problème 1 : compter, ce n'est pas si simple}

\begin{exo}{Réponses}
\begin{enumerate}
  \item \textbf{6 machines} : permutations de 6 objets, $P_6 = 6! = \mathbf{720}$ ordres de tournée.
  \item \textbf{10 machines} : $10! = \mathbf{3\,628\,800}$. Les estimations orales tournent en général autour de quelques centaines ou quelques milliers. L'écart avec la réalité est le premier choc de la séance.
  \item \textbf{Vannes} : chaque vanne a 2 états, les choix sont indépendants, c'est une $p$-liste. \\
  4 vannes : $2^4 = \mathbf{16}$ configurations. \\
  20 vannes : $2^{20} = \mathbf{1\,048\,576}$ configurations.
  \item \textbf{Durée d'essai exhaustif} : $1\,048\,576$ minutes $= 17\,476$ heures $\approx \mathbf{2\,497}$ jours d'essai de 7 heures, soit plus de \textbf{11 ans} à 220 jours travaillés par an.
  \item \textbf{Conclusion} : l'exhaustif est hors de portée dès qu'un système a quelques dizaines de composants. On ne teste pas tout, on \textbf{modélise} et on \textbf{hiérarchise} : c'est exactement la démarche de l'AMDEC (criticité) et celle des probabilités.
\end{enumerate}

\textit{Bonus} : nombre de paires de vannes parmi 20, l'ordre ne compte pas : $C_{20}^{2} = \dfrac{20 \times 19}{2} = \mathbf{190}$.
\end{exo}

\begin{piege}
\textbf{Erreurs attendues.}
\begin{itemize}
  \item Confondre $6!$ et $6^6$ : demander «~une machine peut-elle être visitée deux fois ?~»
  \item Répondre $20 \times 2 = 40$ à la question des 20 vannes : faire dessiner l'arbre des 3 premières vannes au tableau, le doublement apparaît seul.
  \item Répondre $A_{20}^{2} = 380$ au bonus : demander si «~vanne 3 et vanne 7~» et «~vanne 7 et vanne 3~» sont deux défaillances doubles différentes.
\end{itemize}
\end{piege}

\begin{cle}
\textbf{À retenir, à écrire au tableau.} Trois questions à se poser avant de compter : \textbf{y a-t-il répétition ?} \textbf{l'ordre compte-t-il ?} \textbf{prend-on tout ou une partie ?} Elles donnent $n^p$, $A_n^p$, $C_n^p$.
\end{cle}

\section{Problème 2 : le bon sens suffit-il ?}

\begin{exo}{Réponses}
Univers : les prélèvements de 2 roulements parmi 12, sans ordre, en situation d'équiprobabilité. Formule de Laplace.
\begin{enumerate}
  \item $\mathrm{Card}(\Omega) = C_{12}^{2} = \dfrac{12 \times 11}{2} = \mathbf{66}$ prélèvements possibles.
  \item Prélèvements sans aucun défectueux : on choisit 2 roulements parmi les 9 bons, $C_{9}^{2} = \dfrac{9 \times 8}{2} = \mathbf{36}$.
  \item $P(\text{aucun défectueux}) = \dfrac{36}{66} = \dfrac{6}{11} \approx 0{,}5455$. \\[3pt]
  Donc $P(\text{au moins un défectueux}) = 1 - \dfrac{6}{11} = \dfrac{5}{11} \approx \mathbf{0{,}4545}$, soit \textbf{45,5\,\%}.
  \item Environ \textbf{45 pompes sur 100} partiraient avec au moins un roulement défectueux.
  \item Avec 3 roulements : $P(\text{aucun}) = \dfrac{C_{9}^{3}}{C_{12}^{3}} = \dfrac{84}{220} = \dfrac{21}{55} \approx 0{,}3818$, donc \\[3pt]
  $P(\text{au moins un}) = \dfrac{34}{55} \approx \mathbf{0{,}6182}$, soit \textbf{61,8\,\%}. La probabilité grimpe vite : un lot à 25\,\% de défauts contamine presque deux montages sur trois.
\end{enumerate}
\end{exo}

\begin{piege}
\textbf{Les deux raisonnements faux de l'énoncé.}
\begin{itemize}
  \item «~3 sur 12, donc 25\,\%~» : c'est la probabilité qu'\textbf{un seul} roulement tiré soit défectueux. On en tire deux, donc deux occasions de se tromper. Le risque est nécessairement plus élevé.
  \item «~$\frac{3}{12} + \frac{3}{12} = 50\,\%$~» : additionner deux probabilités revient à compter deux fois le cas où les \textbf{deux} roulements sont défectueux. Cette addition n'est légitime que pour des événements incompatibles. Contre-exemple imparable à donner au tableau : avec 5 tirages, la même méthode donnerait $5 \times \frac{3}{12} = 125\,\%$.
\end{itemize}
\end{piege}

\begin{cle}
\textbf{Réflexe à installer.} Dès qu'un énoncé contient «~\textbf{au moins un}~», on calcule l'événement contraire «~aucun~», puis $P(A) = 1 - P(\overline{A})$. Ce réflexe resservira tout le module, notamment avec la loi binomiale.
\end{cle}

\section{Problème 3 : quand l'intuition se trompe lourdement}

\begin{exo}{Réponses}
Notations : $D$ = «~le moteur se dégrade~», $A$ = «~le capteur déclenche l'alarme~». \\
Données : $P(D) = 0{,}01$ \quad ; \quad $P_D(A) = 0{,}95$ \quad ; \quad $P_{\overline{D}}(A) = 0{,}05$.

\begin{enumerate}
  \item Tableau sur 10\,000 moteurs :
  \begin{center}
  \renewcommand{\arraystretch}{1.5}
  \begin{tabular}{|>{\columncolor{uimmlightblue}}l|c|c|c|}
  \hline
  \rowcolor{uimmlightblue}
  \textbf{Sur 10\,000 moteurs} & \textbf{Alarme} & \textbf{Pas d'alarme} & \textbf{Total} \\
  \hline
  Moteur en cours de dégradation & \textbf{95} & 5 & 100 \\
  \hline
  Moteur sain & \textbf{495} & 9\,405 & 9\,900 \\
  \hline
  \textbf{Total} & \textbf{590} & 9\,410 & \textbf{10\,000} \\
  \hline
  \end{tabular}
  \end{center}
  \item Sur les 590 moteurs qui déclenchent une alarme, seuls 95 se dégradent réellement :
  \[ \frac{95}{590} \approx \mathbf{0{,}161} \qquad \text{soit environ } \mathbf{16\,\%}. \]
  Les estimations spontanées du groupe tournent autour de 90 à 95\,\%. L'écart est le point culminant de la séance : \textbf{plus de 8 alarmes sur 10 sont des fausses alertes}, avec un capteur pourtant très performant.
  \item On a calculé $P_A(D)$, la probabilité de dégradation \textbf{sachant que l'alarme sonne}. L'autre, $P_D(A) = 0{,}95$, est la performance du capteur, donnée par le constructeur. Les deux sont très différentes et c'est là que se loge l'erreur de raisonnement. \\[4pt]
  Par la formule des probabilités totales :
  \[ P(A) = P(D)\,P_D(A) + P(\overline{D})\,P_{\overline{D}}(A) = 0{,}01 \times 0{,}95 + 0{,}99 \times 0{,}05 = 0{,}059 \]
  \[ P_A(D) = \frac{P(A \cap D)}{P(A)} = \frac{0{,}01 \times 0{,}95}{0{,}059} = \frac{0{,}0095}{0{,}059} \approx 0{,}1610 \]
  \item Sur les moteurs de plus de 8 ans, $P(D) = 0{,}20$, capteur inchangé :
  \[ P(A) = 0{,}20 \times 0{,}95 + 0{,}80 \times 0{,}05 = 0{,}19 + 0{,}04 = 0{,}23 \]
  \[ P_A(D) = \frac{0{,}19}{0{,}23} \approx \mathbf{0{,}826} \qquad \text{soit environ } \mathbf{83\,\%}. \]
  \item \textbf{Conclusion}. Le capteur n'a pas changé d'un iota. En le posant sur une population où la panne est fréquente, la fiabilité de l'alarme passe de 16\,\% à 83\,\%. \textbf{La valeur d'une alarme dépend autant de la machine qu'on surveille que du capteur qu'on achète.} C'est un argument industriel direct : cibler le parc avant d'investir dans un capteur plus précis.
\end{enumerate}
\end{exo}

\begin{piege}
\textbf{L'erreur centrale : l'oubli du taux de base.} Le groupe retient «~95\,\% de détection~» et oublie que les moteurs sains sont 99 fois plus nombreux. $5\,\%$ d'un très grand nombre écrase $95\,\%$ d'un très petit.

\textbf{Relance si le groupe bloque :} «~Sur 10\,000 moteurs, combien se dégradent ? Combien sont sains ? Maintenant compte les alarmes de chaque côté.~» Le tableau d'effectifs débloque presque toujours, là où l'arbre de probabilités reste abstrait.

\textbf{Relance pour les anciens de première année :} leur faire écrire l'arbre pondéré et retrouver le résultat par Bayes pendant que les autres finissent le tableau.
\end{piege}

\begin{cle}
\textbf{À retenir.} $P_D(A) \neq P_A(D)$. Confondre les deux, c'est confondre «~le capteur détecte les pannes~» et «~l'alarme signale une panne~». La formule de Bayes sert précisément à passer de l'une à l'autre, et elle exige de connaître $P(D)$, la fréquence de la panne dans le parc.
\end{cle}

\vspace{6pt}
\begin{consigne}
\textbf{Transition vers le cours.} Les trois problèmes ouvrent respectivement sur : le dénombrement ($n^p$, $A_n^p$, $C_n^p$), la formule de Laplace et l'événement contraire, l'arbre pondéré avec probabilités totales et Bayes. C'est le plan de la séquence [S01].
\end{consigne}
