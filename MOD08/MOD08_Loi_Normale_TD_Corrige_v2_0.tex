\section{Problème 1 -- Reprise ou rebut}

\needspace{5cm}
\begin{correction}[-- Question 1]
\source{\LN{}, §3.2 Calcul d'une probabilité p.~6 et §3.3 Standardisation p.~7}
$D \sim \mathcal{N}(40{,}005\,;0{,}010)$. Les bornes correspondent à des valeurs centrées réduites entières~:
\[
\frac{39{,}975 - 40{,}005}{0{,}010} = -3, \qquad \frac{40{,}025 - 40{,}005}{0{,}010} = 2.
\]
\[
P(S) = P(D < 39{,}975) = \Phi(-3) \approx 0{,}0013, \qquad P(R) = P(D > 40{,}025) = 1 - \Phi(2) \approx 0{,}0228,
\]
\[
P(C) = P(39{,}975 \leq D \leq 40{,}025) \approx 0{,}9759.
\]
Cohérence~: $C$, $R$, $S$ forment une partition, et $0{,}9759 + 0{,}0228 + 0{,}0013 = 1$.
\piege{oublier que le procédé est décentré et calculer $2\,P(Z > 2{,}5)$~; confondre reprise et rebut.}
\end{correction}

\needspace{4cm}
\begin{correction}[-- Question 2]
\source{\VM{} (probabilité conditionnelle)}
L'évènement «~bague écartée~» est $\overline{C} = R \cup S$. Comme $R \subset \overline{C}$~:
\[
P_{\overline{C}}(R) = \frac{P(R)}{P(R) + P(S)} = \frac{0{,}02275}{0{,}02410} \approx 0{,}9440.
\]
$94\,\%$ des bagues écartées sont récupérables~: le décentrage vers le haut protège contre le rebut.
\piege{répondre $P(R) = 0{,}0228$ sans conditionner.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3]
\source{\LD{}, §1.2 Loi de probabilité et §1.3 Espérance p.~3}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|l|c|c|c|}\hline
$k$ (\texteuro) & $0$ & $1{,}50$ & $9{,}00$ \\ \hline
$P(K = k)$ & $0{,}9759$ & $0{,}0228$ & $0{,}0013$ \\ \hline
\end{tabular}
\end{center}
\[
E(K) = 0 \times 0{,}9759 + 1{,}50 \times 0{,}02275 + 9{,}00 \times 0{,}00135 \approx 0{,}0463 \text{ \texteuro}.
\]
Sur un grand nombre de bagues, la non-qualité coûte en moyenne $4{,}6$ centimes par bague. Sur un mois~: $12\,000 \times 0{,}0463 \approx 555$ \texteuro.
\piege{calculer une moyenne simple des trois coûts~; oublier la valeur $0$ dans la loi de $K$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 4]
\source{\LN{}, §3.2 p.~6~; \LD{}, §1.3 p.~3}
Pour $\mu = 40{,}000$, les bornes sont à $\pm 2{,}5$ écarts-types~: $P(R) = P(S) = 1 - \Phi(2{,}5) \approx 0{,}0062$.
\[
E(K) = (1{,}50 + 9{,}00) \times 0{,}00621 \approx 0{,}0652 \text{ \texteuro}, \quad\text{soit environ } 782 \text{ \texteuro{} par mois.}
\]
Le centrage augmente le coût de $41\,\%$. Il minimise le nombre de bagues écartées ($1{,}24\,\%$ contre $2{,}41\,\%$), mais pas le coût~: un rebut coûte six fois plus qu'une reprise. Quand les coûts sont dissymétriques, on règle «~côté matière~», du côté où l'erreur se rattrape.
\piege{conclure que le centrage est meilleur parce que le taux de conformes augmente.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 5]
\source{\LN{}, §4.2 Retrouver un paramètre $\mu$ ou $\sigma$ p.~8 (méthode)}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $P(D < 39{,}975) \leq 0{,}0005$.
  \item $P\!\left(Z < \dfrac{39{,}975 - \mu}{0{,}010}\right) \leq 0{,}0005 \iff \dfrac{39{,}975 - \mu}{0{,}010} \leq z_{0{,}0005} \approx -3{,}2905 \iff \mu \geq 39{,}975 + 0{,}032905 = 40{,}007905$.

  Arrondi au micromètre \textbf{par excès}~: $\mu = 40{,}008$ mm. (Avec $40{,}007$ on obtiendrait environ $690$ ppm~: l'exigence ne serait pas tenue.)
  \item Pour $\mu = 40{,}008$~: $P(R) \approx 0{,}0446$, $P(S) \approx 0{,}00048$ ($483$ ppm), $E(K) \approx 1{,}50 \times 0{,}04457 + 9 \times 0{,}00048 \approx 0{,}0712$ \texteuro.

  Le coût moyen passe de $0{,}0463$ à $0{,}0712$ \texteuro{} ($+54\,\%$, environ $854$ \texteuro{} par mois). L'exigence n'est pas neutre~: elle se paie en reprises. C'est un argument de négociation chiffré, ou une incitation à réduire $\sigma$.
\end{enumerate}
\piege{arrondir $\mu$ au plus proche (40,008 ici par chance, mais la démarche doit être justifiée)~; écrire $\geq$ au lieu de $\leq$ après standardisation.}
\end{correction}

\needspace{7cm}
\begin{correction}[-- Question 6]
\source{\LN{}, §3.2 p.~6~; \LD{}, §1.3 p.~3}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|l|*{6}{c|}}\hline
$\mu$ (mm) & $40{,}000$ & $40{,}002$ & $40{,}004$ & $40{,}006$ & $40{,}008$ & $40{,}010$ \\ \hline
$P(R)$ & $0{,}0062$ & $0{,}0107$ & $0{,}0179$ & $0{,}0287$ & $0{,}0446$ & $0{,}0668$ \\ \hline
$P(S)$ (ppm) & $6\,210$ & $3\,467$ & $1\,866$ & $968$ & $483$ & $233$ \\ \hline
$E(K)$ (\texteuro) & $0{,}0652$ & $0{,}0473$ & $0{,}0436$ & $0{,}0518$ & $0{,}0712$ & $0{,}1023$ \\ \hline
\end{tabular}
\end{center}
Sans contrainte, le coût minimal du tableau est atteint vers $\mu = 40{,}004$. Avec l'exigence client ($P(S) \leq 500$ ppm), seuls $40{,}008$ et $40{,}010$ sont admissibles~: on recommande $\mu = 40{,}008$, le moins coûteux des réglages admissibles. Pour faire baisser le coût, il faut réduire $\sigma$~: le réglage seul ne suffit plus.
\end{correction}

\section{Problème 2 -- Deux tours, un seul bac}

\needspace{5cm}
\begin{correction}[-- Question 1]
\source{\LN{}, §3.2 p.~6 et §3.3 p.~7}
Tour $A$, centré~: $P_A(N) = 1 - P(29{,}90 \leq L \leq 30{,}10) = 2\,P\!\left(Z > \tfrac{0{,}10}{0{,}035}\right) \approx 0{,}0043$.

Tour $B$, décentré~: $P(L < 29{,}90) = \Phi(-2{,}8) \approx 0{,}0026$ et $P(L > 30{,}10) = 1 - \Phi(1{,}2) \approx 0{,}1151$, d'où $P_B(N) \approx 0{,}1176$.
\piege{utiliser la formule symétrique pour $B$.}
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 2]
\source{\VM{} (arbre pondéré, formule des probabilités totales)}
Arbre~: premier niveau $A$ ($0{,}65$) et $B$ ($0{,}35$)~; second niveau $N$ ou $\overline{N}$ avec les probabilités de la question 1. $A$ et $B$ forment une partition~:
\[
P(N) = 0{,}65 \times 0{,}004275 + 0{,}35 \times 0{,}117625 \approx 0{,}00278 + 0{,}04117 \approx 0{,}0439.
\]
\end{correction}

\needspace{5cm}
\begin{correction}[-- Question 3]
\source{\VM{} (probabilité conditionnelle «~inversée~»)}
\[
P_N(B) = \frac{P(B \cap N)}{P(N)} = \frac{0{,}04117}{0{,}04395} \approx 0{,}9368.
\]
Le tour $B$ fait $35\,\%$ de la production mais $94\,\%$ des non-conformes. La révision de $B$ est prioritaire~: décentrage de $0{,}04$ mm et dispersion plus forte.
\piege{répondre $P_B(N) = 0{,}1176$~; confondre «~part de la production~» et «~part des défauts~».}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 4]
\source{\LN{}, §3.2 p.~6~; \VM{} (formule de Bayes)}
$P_A(H) = P(30{,}05 < L \leq 30{,}10) \approx 0{,}0744$ pour le tour $A$~; $P_B(H) \approx 0{,}3057$ pour le tour $B$.
\[
P(H) = 0{,}65 \times 0{,}07443 + 0{,}35 \times 0{,}30567 \approx 0{,}1554, \qquad
P_H(B) = \frac{0{,}35 \times 0{,}30567}{0{,}1554} \approx 0{,}6886.
\]
Une pièce en zone d'alerte vient de $B$ avec une probabilité de $69\,\%$, contre $35\,\%$ sans information. Une mesure même conforme renseigne sur l'origine.
\end{correction}

\needspace{7cm}
\begin{correction}[-- Question 5]
\source{\VM{} (probabilités totales)~; \LN{}, §4.2 p.~8 et §2.3 Problème inverse p.~5}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $P(N) = 0{,}65\,P_A(N) + 0{,}35\,P_B(N) \leq 0{,}01 \iff P_B(N) \leq \dfrac{0{,}01 - 0{,}65 \times 0{,}004275}{0{,}35} \approx 0{,}0206$.
  \item $B$ est centré~: $P_B(N) = 2\,P\!\left(Z > \tfrac{0{,}10}{\sigma_B}\right) \leq 0{,}020633 \iff \Phi\!\left(\tfrac{0{,}10}{\sigma_B}\right) \geq 0{,}98968 \iff \dfrac{0{,}10}{\sigma_B} \geq 2{,}3146$.

  D'où $\sigma_B \leq 0{,}0432$ mm, soit $\sigma_B = 0{,}043$ mm arrondi \textbf{par défaut}. Le recentrage seul ne suffit pas~: avec $\sigma_B = 0{,}05$, on aurait $P_B(N) \approx 4{,}6\,\%$.
\end{enumerate}
\piege{oublier le facteur 2 (deux queues)~; inverser le sens de l'inégalité en isolant $\sigma_B$.}
\end{correction}

\section{Problème 3 -- Remplissage, audit client et investissement}

\needspace{3.5cm}
\begin{correction}[-- Question 1]
\source{\LN{}, §3.2 p.~6}
$p = P(V < 5{,}00) = \Phi\!\left(\tfrac{5{,}00 - 5{,}05}{0{,}03}\right) = \Phi(-1{,}667) \approx 0{,}0478$.
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 2]
\source{\LD{}, §2.2 Schéma de Bernoulli p.~5, §3.1 Définition et §3.3 Calcul des probabilités p.~7}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item Épreuve~: examiner un bidon~; succès «~sous-rempli~», probabilité $p$. $12$ répétitions identiques et indépendantes. $X$ compte les succès~: $X \sim \mathcal{B}(12\,;0{,}0478)$.
  \item $P(X = 0) = (1 - p)^{12} \approx 0{,}5556$~; $P(X = 1) \approx 0{,}3346$~; $P(X \geq 2) = 1 - P(X=0) - P(X=1) \approx 0{,}1097$.
  \item $P(\text{refus}) = P(X \geq 1) = 1 - P(X = 0) \approx 0{,}4444$. Près d'un carton sur deux est refusé~: le réglage à $5{,}05$ L, qui paraît généreux, est inacceptable dès qu'on raisonne au niveau du carton.
\end{enumerate}
\piege{calculer $P(X \geq 2)$ comme $1 - P(X \leq 2)$~; oublier l'hypothèse d'indépendance.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 3]
\source{\LD{}, §3.3 p.~7~; \LN{}, §4.2 p.~8}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $1 - (1-p)^{12} \leq 0{,}02 \iff (1-p)^{12} \geq 0{,}98 \iff 1 - p \geq 0{,}98^{1/12}$ (la fonction $x \mapsto x^{1/12}$ est croissante sur $[0\,;+\infty[$) $\iff p \leq 1 - 0{,}98^{1/12} \approx 0{,}001682$.
  \item $\Phi\!\left(\tfrac{5 - \mu}{0{,}03}\right) \leq 0{,}001682 \iff \dfrac{5 - \mu}{0{,}03} \leq z_{0{,}001682} \approx -2{,}9323 \iff \mu \geq 5{,}0880$.

  Arrondi au millilitre \textbf{par excès}~: $\mu = 5{,}088$ L.
\end{enumerate}
\piege{imposer $p \leq 0{,}02$ (confusion bidon / carton)~; arrondir $\mu$ par défaut.}
\end{correction}

\needspace{4cm}
\begin{correction}[-- Question 4]
\source{\LN{}, §3.1 rôle des paramètres p.~6 ($\mu$ est le volume moyen)}
Volume donné par an~: $0{,}088 \times 150\,000 = 13\,200$ L. Coût~: $13\,200 \times 3{,}20 = 42\,240$ \texteuro{} par an.
\piege{comparer au coût du réglage actuel ($24\,000$ \texteuro) comme s'il était une option~: ce réglage ne respecte pas l'exigence du client.}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 5]
\source{\LN{}, §4.2 p.~8}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item Le majorant de $p$ ne dépend pas de $\sigma$~: $\mu \geq 5 + 2{,}9323 \times 0{,}012 \approx 5{,}0352$, soit $\mu = 5{,}036$ L.
  \item Surremplissage~: $0{,}036 \times 150\,000 = 5\,400$ L, soit $17\,280$ \texteuro. Économie~: $42\,240 - 17\,280 = 24\,960$ \texteuro{} par an.
  \item $\dfrac{18\,000}{24\,960} \times 12 \approx 8{,}7$~: l'investissement est remboursé en moins de $9$ mois. Réduire la dispersion permet de rapprocher le réglage de la cote~: c'est là que se trouve le gain.
\end{enumerate}
\end{correction}

\section{Problème 4 -- Jeu d'assemblage et empilement de cotes}

\needspace{4cm}
\begin{correction}[-- Question 1]
\source{\OM{} (résultat admis)}
$A$ et $B$ sont indépendantes~: $J = A - B \sim \mathcal{N}\!\left(0{,}028\,;\sqrt{0{,}006^2 + 0{,}005^2}\right)$, soit $\sigma_J = \sqrt{6{,}1 \times 10^{-5}} \approx 0{,}00781$ mm.

L'affirmation est fausse~: les dispersions ne se compensent pas, elles se cumulent. Les \textbf{variances} s'ajoutent, même pour une différence, et $\sigma_J$ est supérieur à $\sigma_A$ et à $\sigma_B$.
\end{correction}

\needspace{4cm}
\begin{correction}[-- Question 2]
\source{\LN{}, §3.2 p.~6}
$P(0{,}010 \leq J \leq 0{,}050) \approx 0{,}9870$~; $P(J < 0{,}010) \approx 0{,}0106$~; $P(J > 0{,}050) \approx 0{,}0024$.
\end{correction}

\needspace{4cm}
\begin{correction}[-- Question 3]
\source{\VM{} (probabilité conditionnelle)}
\[
P_{\text{NF}}(J < 0{,}010) = \frac{0{,}010593}{0{,}010593 + 0{,}002425} \approx 0{,}8137.
\]
Le jeu moyen ($0{,}028$) est plus proche de la borne basse~: $81\,\%$ des refus sont des jeux insuffisants.
\end{correction}

\needspace{4cm}
\begin{correction}[-- Question 4]
\source{\LN{}, §3.2 p.~6~; \LD{}, §3, espérance d'une loi binomiale p.~8}
$P(J < 0) \approx 0{,}000169$, soit environ $169$ ppm. Sur $50\,000$ montages~: $50\,000 \times 0{,}000169 \approx 8{,}4$, soit environ $8$ montages impossibles par an.
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 5]
\source{\LN{}, §4.2 p.~8}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $\dfrac{0{,}010 - \mu_J}{0{,}00781} \leq z_{0{,}001} \approx -3{,}0902 \iff \mu_J \geq 0{,}03414$. Or $\mu_J = \mu_A - 19{,}997$, donc $\mu_A \geq 20{,}03114$, soit $\mu_A = 20{,}032$ mm par excès.
  \item Avec $\mu_J = 0{,}035$~: $P(J < 0{,}010) \approx 0{,}0007$ et $P(J > 0{,}050) \approx 0{,}0274$. Non fonctionnels~: $2{,}81\,\%$ contre $1{,}30\,\%$ au départ.

  Décaler le réglage transfère les défauts d'une borne à l'autre et, ici, en augmente le total. Le problème est la dispersion de $J$, pas son centrage.
\end{enumerate}
\end{correction}

\needspace{6cm}
\begin{correction}[-- Question 6]
\source{\OM{}~; \LN{}, §2.3 p.~5 et §4.2 p.~8}
$J$ centré sur $0{,}030$~: $2\,P\!\left(Z > \tfrac{0{,}020}{\sigma_J}\right) \leq 0{,}005 \iff \dfrac{0{,}020}{\sigma_J} \geq z_{0{,}9975} \approx 2{,}8070 \iff \sigma_J \leq 0{,}007125$.

\begin{correction}[-- Question 2]
\source{\OM{}~; \LE{}, §4.1 La propriété d'absence de mémoire p.~7}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $P_{T > 7000}(T > 10\,000) = \dfrac{P(T > 10\,000)}{P(T > 7\,000)} = \dfrac{0{,}25249}{0{,}90879} \approx 0{,}2778$.
  \item $P(T > 3\,000) = 1 - \Phi(-4) \approx 1{,}0000$. Une pompe neuve tient presque sûrement $3\,000$ h~; une pompe de $7\,000$ h n'a que $28\,\%$ de chances de tenir $3\,000$ h de plus. Elle a vieilli.
  \item Avec une loi exponentielle, $P_{T > 7000}(T > 10\,000) = P(T > 3\,000)$~: le composant «~oublie~» son âge. Ce modèle convient aux défaillances aléatoires (électronique), pas à l'usure. Pour une pièce d'usure, la loi normale est plus pertinente.
\end{enumerate}
\piege{calculer $P(T > 10\,000)$ sans conditionner~; appliquer l'absence de mémoire à la loi normale.}
\end{correction}

\begin{correction}[-- Question 4]
\source{\VM{} (probabilité conditionnelle)~; \LN{}, §4.1 p.~8}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $P_{T > 7000}(T \leq 7\,000 + d) \leq 0{,}10$.
  \item Cette probabilité vaut $1 - \dfrac{P(T > 7\,000 + d)}{P(T > 7\,000)}$. La condition s'écrit $\dfrac{P(T > 7000 + d)}{P(T > 7000)} \geq 0{,}9$, d'où le résultat ($P(T > 7000) > 0$).
  \item $P(T > 7\,000 + d) \geq 0{,}9 \times 0{,}908789 \approx 0{,}81791$, soit $P(T \leq 7\,000 + d) \leq 0{,}18209$.

  $7\,000 + d \leq \texttt{FracNormale}(0{,}18209\,;9000\,;1500) \approx 7\,638{,}9$, d'où $d \leq 638{,}9$~: $d = 638$ h par défaut.
\end{enumerate}
Remarque~: cette pompe a déjà dépassé l'âge de remplacement préventif $t_0$. La question 4 chiffre le risque qu'on accepte en la laissant tourner.
\piege{résoudre $P(T \leq 7000 + d) = 0{,}10$ sans conditionner (on trouverait $7000 + d \approx 7078$, soit $d \approx 78$ h, par un raisonnement faux).}

\needspace{6cm}
\begin{correction}[-- Question 1]
\source{\LN{}, §3.2 p.~6~; \VM{} (arbre pondéré, probabilités totales)}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $\alpha = P(\Delta > 3{,}5)$ avec $\Delta \sim \mathcal{N}(2\,;0{,}5)$~: $\alpha = 1 - \Phi(3) \approx 0{,}0013$.

  $\beta = P(\Delta \leq 3{,}5)$ avec $\Delta \sim \mathcal{N}(5\,;1)$~: $\beta = \Phi(-1{,}5) \approx 0{,}0668$.
  \item Arbre~: $E$ ($0{,}96$) puis $R$ ($\alpha$) / $\overline{R}$~; $F$ ($0{,}04$) puis $R$ ($1 - \beta$) / $\overline{R}$ ($\beta$).
  \[
  P(R) = 0{,}96 \times 0{,}00135 + 0{,}04 \times 0{,}93319 \approx 0{,}00130 + 0{,}03733 \approx 0{,}0386.
  \]
\end{enumerate}

\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item $\beta = \Phi(s - 5) \leq 0{,}005 \iff s - 5 \leq z_{0{,}005} \approx -2{,}5758 \iff s \leq 2{,}4242$. Arrondi \textbf{par défaut}~: $s = 2{,}42$ mbar ($\beta$ croît avec $s$).

  \item \begin{center}

  \begin{tabular}{|l|c|c|c|c|c|}\hline

\begin{correction}[-- Question 4]
\source{\VM{} (indépendance, arbre à deux niveaux)}
\begin{enumerate}[label=\alph*), leftmargin=14pt, itemsep=2pt]
  \item Carter étanche rebuté~: rejeté deux fois, $\alpha^2 \approx 0{,}2005^2 \approx 0{,}0402$. Carter fuyard rebuté~: $(1 - \beta)^2 \approx 0{,}9901$.

Alors $\alpha \approx 0{,}352$ et $\alpha^2 \approx 0{,}124$~: on rebute $12\,\%$ des étanches au lieu de $20\,\%$, en respectant l'exigence. Le retest reste intéressant, à condition de resserrer le seuil.
\end{correction}
