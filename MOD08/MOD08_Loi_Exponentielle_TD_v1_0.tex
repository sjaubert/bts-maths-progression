\begin{consigne}
\textbf{Consignes.} Les probabilités sont arrondies à $10^{-4}$, les durées à l'unité près. Pour $T \sim \mathcal{E}(\lambda)$, on rappelle $R(t) = P(T>t) = e^{-\lambda t}$, $F(t) = 1 - e^{-\lambda t}$ et $E(T) = 1/\lambda$ (MTBF). \textbf{Vérifier la cohérence des unités} entre $\lambda$ et $t$. Calculatrice autorisée : calcul direct avec la touche \texttt{$e^x$} (pas de menu dédié).
\end{consigne}

\section{Calculs de base \hfill {\normalsize\color{uimmred}[C04]}}

\begin{exo}{Exercice 1 -- Durée de vie d'un relais}
La durée de vie $T$ (en heures) d'un relais suit $\mathcal{E}(\lambda)$ avec $\lambda = 0{,}0005$ h$^{-1}$.
\begin{enumerate}[label=\alph*)]
  \item Donner l'expression de la fonction de survie $R(t)$.
  \item Calculer $P(T > 2000)$ et $P(T \leq 1500)$.
  \item Calculer $P(1000 \leq T \leq 3000)$.
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 2 -- Lecture d'un taux}
Un variateur a un taux de défaillance $\lambda = 0{,}00008$ h$^{-1}$.
\begin{enumerate}[label=\alph*)]
  \item Calculer la probabilité qu'il fonctionne au-delà de $10\,000$ h.
  \item Calculer la probabilité qu'il tombe en panne avant $5000$ h.
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 3 -- Médiane de durée de vie}
La durée de vie $T$ (en heures) d'une LED de puissance suit $\mathcal{E}(0{,}00002)$.
\begin{enumerate}[label=\alph*)]
  \item Calculer $R(20\,000)$.
  \item Déterminer la durée $t_m$ telle que $P(T > t_m) = 0{,}5$ (durée de vie médiane).
\end{enumerate}
\end{exo}

\section{MTBF, espérance et écart-type \hfill {\normalsize\color{uimmred}[C05]}}

\begin{exo}{Exercice 4 -- MTBF d'un moteur}
La durée de vie d'un moteur suit une loi exponentielle de taux $\lambda = 0{,}00025$ h$^{-1}$.
\begin{enumerate}[label=\alph*)]
  \item Calculer la MTBF et l'écart-type de la durée de vie.
  \item Calculer la probabilité que le moteur dépasse sa MTBF.
  \item Ce résultat dépend-il de la valeur de $\lambda$ ? Justifier.
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 5 -- Du besoin à la MTBF}
Un bureau d'études souhaite qu'une pompe ait une durée de vie moyenne d'au moins $12\,000$ h.
\begin{enumerate}[label=\alph*)]
  \item Quel taux de défaillance maximal $\lambda$ doit respecter la pompe ?
  \item Avec ce taux, calculer la fiabilité à $6000$ h.
\end{enumerate}
\end{exo}

\section{Durée de vie sans vieillissement \hfill {\normalsize\color{uimmred}[C05]}}

\begin{exo}{Exercice 6 -- Absence de mémoire}
La durée de vie $T$ d'un capteur suit $\mathcal{E}(0{,}0004)$ (en heures). Le capteur est en service depuis $3000$ h et fonctionne encore.
\begin{enumerate}[label=\alph*)]
  \item Calculer la probabilité qu'il fonctionne encore au moins $1000$ h de plus.
  \item Comparer avec la probabilité qu'un capteur neuf tienne $1000$ h.
  \item Quel principe cette égalité illustre-t-elle ? En quoi limite-t-il le modèle pour des pièces soumises à l'usure ?
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 7 -- Un composant « vieilli » ?}
Un fournisseur affirme qu'un de ses fusibles thermiques, de durée de vie exponentielle, est « plus fiable après rodage ». Un technicien objecte.
\begin{enumerate}[label=\alph*)]
  \item Que dit la loi exponentielle sur l'effet de l'âge du composant ?
  \item Le discours du fournisseur est-il compatible avec le modèle exponentiel ? Justifier.
\end{enumerate}
\end{exo}

\section{Fiabilité et maintenance \hfill {\normalsize\color{uimmred}[C04]\,[C05]}}

\begin{exo}{Exercice 8 -- Périodicité préventive}
Un roulement a un taux de défaillance $\lambda = 0{,}0003$ h$^{-1}$.
\begin{enumerate}[label=\alph*)]
  \item Exprimer la fiabilité $R(t)$.
  \item Déterminer la durée $t$ au bout de laquelle la fiabilité tombe à $0{,}90$.
  \item On veut maintenir une fiabilité d'au moins $0{,}95$ : quelle périodicité de remplacement préconiser ?
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 9 -- De la MTBF à la garantie}
Une carte électronique a une MTBF de $8000$ h (durée de vie exponentielle).
\begin{enumerate}[label=\alph*)]
  \item Déterminer le taux de défaillance $\lambda$.
  \item Calculer la fiabilité à $4000$ h et à $8000$ h.
  \item Déterminer la durée de garantie $t_0$ telle que $95\,\%$ des cartes fonctionnent encore.
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 10 -- Fonctionnement sur une mission}
Un équipement embarqué doit fonctionner sans panne pendant une mission de $600$ h. Sa durée de vie suit $\mathcal{E}(\lambda)$ avec $\lambda = 0{,}0002$ h$^{-1}$.
\begin{enumerate}[label=\alph*)]
  \item Calculer la probabilité de réussir la mission (aucune panne sur $600$ h).
  \item Quelle MTBF faudrait-il pour porter cette probabilité à $0{,}99$ ?
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 11 -- Deux composants en série}
Deux composants indépendants, de durées de vie exponentielles de taux $\lambda_1 = 0{,}0002$ h$^{-1}$ et $\lambda_2 = 0{,}0003$ h$^{-1}$, sont montés \textbf{en série} : le système tombe en panne dès que l'un des deux lâche.
\begin{enumerate}[label=\alph*)]
  \item Exprimer la fiabilité du système $R(t) = R_1(t) \times R_2(t)$.
  \item Montrer que le système suit encore une loi exponentielle et donner son taux.
  \item Calculer sa MTBF, puis $R(1000)$.
\end{enumerate}
\end{exo}

\section{Lien avec Poisson et simulation \hfill {\normalsize\color{uimmred}[C04]}}

\begin{exo}{Exercice 12 -- Temps entre deux pièces}
Un poste reçoit en moyenne $4$ pièces par minute, selon un processus de Poisson.
\begin{enumerate}[label=\alph*)]
  \item Donner la loi du temps d'attente $T$ (en minutes) entre deux pièces et sa moyenne.
  \item Calculer la probabilité d'attendre plus de $20$ secondes.
  \item Calculer la probabilité d'attendre moins de $10$ secondes.
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 13 -- Cohérence Poisson / exponentielle}
Une machine tombe en panne en moyenne $1{,}5$ fois par an, selon un processus de Poisson.
\begin{enumerate}[label=\alph*)]
  \item Calculer, avec la loi de Poisson, la probabilité qu'il n'y ait aucune panne sur une année.
  \item Retrouver ce résultat avec la loi exponentielle du temps d'attente $T$ (en années).
  \item Calculer le temps d'attente moyen entre deux pannes.
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 14 -- Simulation}
On veut simuler des durées de vie suivant $\mathcal{E}(\lambda)$ avec $\lambda = 0{,}0004$ h$^{-1}$, à partir de tirages $U$ uniformes sur $[0\,;1]$.
\begin{enumerate}[label=\alph*)]
  \item Rappeler la formule donnant $T$ en fonction de $U$ et $\lambda$.
  \item Pour $U = 0{,}5$, calculer la durée $T$ obtenue. Comparer à la durée de vie médiane.
  \item Écrire un algorithme (langage naturel ou Python) estimant la MTBF sur $N$ simulations.
\end{enumerate}
\end{exo}

\begin{exo}{Exercice 15 -- Synthèse : politique de remplacement}
Un servomoteur a une durée de vie exponentielle de MTBF $10\,000$ h.
\begin{enumerate}[label=\alph*)]
  \item Déterminer $\lambda$, puis calculer $R(5000)$ et $F(2000)$.
  \item Déterminer la durée $t_0$ garantissant une fiabilité de $0{,}98$.
  \item Un parc contient $50$ servomoteurs indépendants. En admettant la fiabilité de la question a) à $5000$ h, quel est le nombre moyen de moteurs encore en service à $5000$ h ?
  \item Le seuil d'alerte est atteint quand la fiabilité individuelle passe sous $0{,}80$. À partir de quelle durée de fonctionnement est-on dans cette zone ?
\end{enumerate}
\end{exo}
