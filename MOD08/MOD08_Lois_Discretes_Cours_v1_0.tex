\section{Variables aléatoires discrètes \hfill {\normalsize\color{uimmred}[C05]}}

\subsection{Notion de variable aléatoire}

\begin{definition}
Une \textbf{variable aléatoire} $X$ associe un nombre réel à chaque issue d'une expérience aléatoire. Elle est \textbf{discrète} lorsqu'elle ne prend qu'un nombre fini (ou dénombrable) de valeurs isolées $x_1, x_2, \dots, x_n$.
\end{definition}

\begin{exemple}[-- Situations discrètes]
Le nombre de pièces défectueuses dans un lot, le nombre de pannes d'une machine dans un mois, le nombre de retouches sur une série : ces grandeurs se \textbf{comptent}, elles sont discrètes.
\end{exemple}

\subsection{Loi de probabilité}

\begin{definition}
Donner la \textbf{loi de probabilité} de $X$, c'est associer à chaque valeur $x_i$ sa probabilité $p_i = P(X = x_i)$. On la présente dans un tableau :
\end{definition}

\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|}
\hline
\rowcolor{uimmlightblue}
$x_i$ & $x_1$ & $x_2$ & $\cdots$ & $x_n$ \\
\hline
$P(X = x_i)$ & $p_1$ & $p_2$ & $\cdots$ & $p_n$ \\
\hline
\end{tabular}
\end{center}

\begin{theoreme}[-- Condition fondamentale]
La somme de toutes les probabilités d'une loi vaut $1$ :
\[
p_1 + p_2 + \dots + p_n = \sum_{i=1}^{n} P(X = x_i) = 1.
\]
\end{theoreme}

\subsection{Espérance, variance, écart-type \hfill {\normalsize\color{uimmred}[C05]}}

\begin{definition}
Pour une variable aléatoire discrète $X$ de loi $\big(x_i\,;p_i\big)$ :
\[
\underbrace{E(X) = \sum_{i=1}^{n} x_i\,p_i}_{\text{espérance}}, \qquad
\underbrace{V(X) = \sum_{i=1}^{n} x_i^2\,p_i - \big(E(X)\big)^2}_{\text{variance}}, \qquad
\underbrace{\sigma(X) = \sqrt{V(X)}}_{\text{écart-type}}.
\]
\end{definition}

\begin{remarque}[-- Interprétation]
L'\textbf{espérance} $E(X)$ est la valeur moyenne attendue sur un très grand nombre de répétitions. L'\textbf{écart-type} $\sigma(X)$ mesure la dispersion des résultats autour de cette moyenne. En production, $E(X)$ donne le niveau moyen (par exemple le nombre moyen de rebuts) et $\sigma(X)$ sa régularité.
\end{remarque}

\begin{exemple}[-- Nombre de retouches par pièce]
Sur un poste, le nombre $X$ de retouches par pièce suit la loi :
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|c|c|}
\hline
\rowcolor{uimmlightblue}
$x_i$ & $0$ & $1$ & $2$ & $3$ \\
\hline
$p_i$ & $0{,}50$ & $0{,}30$ & $0{,}15$ & $0{,}05$ \\
\hline
\end{tabular}
\end{center}
On vérifie $\sum p_i = 1$. Alors :
\[
E(X) = 0\times0{,}5 + 1\times0{,}3 + 2\times0{,}15 + 3\times0{,}05 = 0{,}75,
\]
\[
V(X) = \big(0^2\times0{,}5 + 1^2\times0{,}3 + 2^2\times0{,}15 + 3^2\times0{,}05\big) - 0{,}75^2 = 1{,}35 - 0{,}5625 = 0{,}7875,
\]
\[
\sigma(X) = \sqrt{0{,}7875} \approx 0{,}89.
\]
En moyenne, $0{,}75$ retouche par pièce.
\end{exemple}

\section{La loi de Bernoulli \hfill {\normalsize\color{uimmred}[C01]}}

\subsection{Épreuve de Bernoulli}

\begin{definition}
Une \textbf{épreuve de Bernoulli} est une expérience aléatoire à \textbf{deux issues} seulement :
\begin{itemize}
  \item le \textbf{succès} $S$, de probabilité $p$ ;
  \item l'\textbf{échec} $\overline{S}$, de probabilité $1 - p$.
\end{itemize}
\end{definition}

\begin{definition}[-- Loi de Bernoulli]
La variable $X$ qui vaut $1$ en cas de succès et $0$ en cas d'échec suit la \textbf{loi de Bernoulli} de paramètre $p$, notée $\mathcal{B}(p)$ :
\[
P(X = 1) = p, \qquad P(X = 0) = 1 - p.
\]
\end{definition}

\begin{theoreme}[-- Espérance et écart-type]
Pour $X \sim \mathcal{B}(p)$ :
\[
E(X) = p, \qquad V(X) = p(1 - p), \qquad \sigma(X) = \sqrt{p(1 - p)}.
\]
\end{theoreme}

\begin{appliindus}[-- Contrôle « conforme / non conforme »]
Prélever une pièce et tester sa conformité est une épreuve de Bernoulli : succès $=$ « pièce conforme ». Si le procédé produit $5\,\%$ de rebut, alors $p = 0{,}95$ pour l'événement « conforme », et $\sigma = \sqrt{0{,}95 \times 0{,}05} \approx 0{,}2179$.
\end{appliindus}

\subsection{Schéma de Bernoulli \hfill {\normalsize\color{uimmred}[C01]}}

\begin{definition}
Un \textbf{schéma de Bernoulli} est la répétition de $n$ épreuves de Bernoulli \textbf{identiques} et \textbf{indépendantes}, chacune de paramètre $p$.
\end{definition}

\begin{remarque}
Trois conditions sont à réunir : \textbf{deux issues} par épreuve, un paramètre $p$ \textbf{constant}, et l'\textbf{indépendance} des épreuves. Le tirage \emph{avec remise} respecte l'indépendance ; le tirage \emph{sans remise} ne la respecte pas (sauf si la population est très grande devant l'échantillon).
\end{remarque}

\begin{lstlisting}[style=uimmpython, caption={Simulation d'un schéma de Bernoulli (n = 20, p = 0.95)}]
import random

n, p = 20, 0.95
resultats = [1 if random.random() < p else 0 for _ in range(n)]
nb_succes = sum(resultats)
print("Tirages :", resultats)
print("Nombre de succes :", nb_succes)
\end{lstlisting}

\section{La loi binomiale \hfill {\normalsize\color{uimmred}[C02]\,[C03]\,[C04]}}

\subsection{Définition}

\begin{definition}
Dans un schéma de Bernoulli de $n$ épreuves de paramètre $p$, la variable $X$ égale au \textbf{nombre de succès} suit la \textbf{loi binomiale} de paramètres $n$ et $p$, notée $\mathcal{B}(n\,;p)$.
\end{definition}

\begin{methode}[-- Reconnaître une loi binomiale {\normalsize\color{uimmred}[C02]}]
On vérifie les quatre points :
\begin{enumerate}
  \item on répète $n$ fois la même épreuve ;
  \item chaque épreuve n'a que \textbf{deux issues} (succès / échec) ;
  \item les épreuves sont \textbf{indépendantes} ;
  \item la probabilité $p$ de succès est \textbf{constante}.
\end{enumerate}
Alors $X = $ « nombre de succès » suit $\mathcal{B}(n\,;p)$.
\end{methode}

\subsection{Coefficient binomial}

\begin{definition}
Le nombre de façons de choisir $k$ succès parmi $n$ épreuves est le \textbf{coefficient binomial} $\dbinom{n}{k}$ (lu « $k$ parmi $n$ ») :
\[
\binom{n}{k} = \frac{n!}{k!\,(n-k)!}.
\]
\end{definition}

\begin{exemple}
$\dbinom{5}{2} = \dfrac{5!}{2!\,3!} = 10$ ; \quad $\dbinom{10}{3} = 120$ ; \quad $\dbinom{n}{0} = \dbinom{n}{n} = 1$.
\end{exemple}

\subsection{Calcul des probabilités \hfill {\normalsize\color{uimmred}[C04]}}

\begin{theoreme}[-- Terme général]
Pour $X \sim \mathcal{B}(n\,;p)$ et $0 \leq k \leq n$ :
\[
\boxed{P(X = k) = \binom{n}{k}\,p^{\,k}\,(1 - p)^{\,n-k}}
\]
\end{theoreme}

\begin{exemple}[-- Calcul détaillé]
Pour $X \sim \mathcal{B}(6\,;0{,}2)$ :
\[
P(X = 2) = \binom{6}{2}\,0{,}2^{2}\,0{,}8^{4} = 15 \times 0{,}04 \times 0{,}4096 \approx 0{,}2458.
\]
\end{exemple}

\begin{calculatrice}[-- Loi binomiale $\mathcal{B}(n\,;p)$]
\begin{itemize}
  \item \textbf{TI (82/83/84)} : \texttt{binomFdp(n, p, k)} pour $P(X = k)$ ; \; \texttt{binomFRép(n, p, k)} pour $P(X \leq k)$.
  \item \textbf{Casio (Graph)} : menu \texttt{STAT} $\rightarrow$ \texttt{DIST} $\rightarrow$ \texttt{BINM} $\rightarrow$ \texttt{Bpd} pour $P(X = k)$ et \texttt{Bcd} pour $P(X \leq k)$.
\end{itemize}
Pour $P(X \geq k)$, utiliser $P(X \geq k) = 1 - P(X \leq k-1)$.
\end{calculatrice}

\begin{theoreme}[-- Espérance et écart-type {\normalsize\color{uimmred}[C05]}]
Pour $X \sim \mathcal{B}(n\,;p)$ :
\[
E(X) = np, \qquad V(X) = np(1 - p), \qquad \sigma(X) = \sqrt{np(1 - p)}.
\]
\end{theoreme}

\subsection{Représentation graphique \hfill {\normalsize\color{uimmred}[C03]}}

On représente la loi binomiale par un \textbf{diagramme en bâtons} : en abscisse le nombre de succès $k$, en ordonnée la probabilité $P(X = k)$.

\begin{appliindus}[-- Contrôle qualité d'un lot]
Un procédé produit $5\,\%$ de pièces défectueuses. On prélève $n = 20$ pièces (population très grande, tirage assimilé à indépendant). Le nombre $X$ de pièces défectueuses suit $\mathcal{B}(20\,;0{,}05)$ : $E(X) = 1$ pièce, $\sigma(X) \approx 0{,}97$.
\begin{itemize}
  \item $P(X = 0) = 0{,}95^{20} \approx 0{,}3585$ (lot parfait) ;
  \item $P(X \leq 2) \approx 0{,}9245$ ;
  \item $P(X \geq 1) = 1 - P(X = 0) \approx 0{,}6415$.
\end{itemize}
\end{appliindus}

\section{La loi de Poisson \hfill {\normalsize\color{uimmred}[C04]\,[C05]}}

\subsection{Définition}

La loi de Poisson modélise le nombre d'\textbf{événements rares} survenant pendant une durée, sur une longueur ou une surface données : pannes par mois, défauts par mètre carré, arrivées par heure.

\begin{definition}
La variable $X$ suit la \textbf{loi de Poisson} de paramètre $\lambda > 0$, notée $\mathcal{P}(\lambda)$, si pour tout entier $k \geq 0$ :
\[
\boxed{P(X = k) = e^{-\lambda}\,\frac{\lambda^{\,k}}{k!}}
\]
Le paramètre $\lambda$ représente le \textbf{nombre moyen} d'événements sur l'intervalle considéré.
\end{definition}

\begin{theoreme}[-- Espérance et écart-type {\normalsize\color{uimmred}[C05]}]
Pour $X \sim \mathcal{P}(\lambda)$ :
\[
E(X) = \lambda, \qquad V(X) = \lambda, \qquad \sigma(X) = \sqrt{\lambda}.
\]
\end{theoreme}

\begin{remarque}
Propriété caractéristique : pour une loi de Poisson, \textbf{l'espérance et la variance sont égales} à $\lambda$. C'est un bon test : si, sur des relevés, moyenne et variance sont proches, le modèle de Poisson est plausible.
\end{remarque}

\begin{calculatrice}[-- Loi de Poisson $\mathcal{P}(\lambda)$]
\begin{itemize}
  \item \textbf{TI} : \texttt{poissonFdp($\lambda$, k)} pour $P(X = k)$ ; \; \texttt{poissonFRép($\lambda$, k)} pour $P(X \leq k)$.
  \item \textbf{Casio} : \texttt{DIST} $\rightarrow$ \texttt{POISN} $\rightarrow$ \texttt{Ppd} pour $P(X = k)$ et \texttt{Pcd} pour $P(X \leq k)$.
\end{itemize}
\end{calculatrice}

\begin{exemple}[-- Calcul détaillé]
Pour $X \sim \mathcal{P}(2)$ :
\[
P(X = 3) = e^{-2}\,\frac{2^{3}}{3!} = e^{-2}\,\frac{8}{6} \approx 0{,}1804.
\]
\end{exemple}

\begin{appliindus}[-- Pannes d'une machine]
Une machine tombe en panne en moyenne $3$ fois par mois. En admettant un modèle de Poisson, le nombre $X$ de pannes mensuelles suit $\mathcal{P}(3)$ :
\begin{itemize}
  \item $P(X = 0) = e^{-3} \approx 0{,}0498$ (mois sans panne) ;
  \item $P(X \leq 2) \approx 0{,}4232$ ;
  \item $P(X \geq 1) = 1 - e^{-3} \approx 0{,}9502$.
\end{itemize}
Ces valeurs alimentent le dimensionnement d'un stock de pièces de rechange.
\end{appliindus}

\newpage
\section{Approximation d'une loi binomiale par une loi de Poisson \hfill {\normalsize\color{uimmred}[C04]}}

\subsection{Principe}

Quand une loi binomiale porte sur un \textbf{grand nombre d'épreuves} ($n$ grand) avec une \textbf{probabilité de succès faible} ($p$ petit), le calcul direct devient lourd et la loi de Poisson en donne une excellente approximation. On parle de \textbf{loi des petits nombres}.

\begin{theoreme}[-- Approximation de Poisson]
Si $X \sim \mathcal{B}(n\,;p)$ avec $n$ grand et $p$ petit, on peut approcher $X$ par la loi de Poisson de même espérance :
\[
\mathcal{B}(n\,;p) \approx \mathcal{P}(\lambda) \quad\text{avec}\quad \lambda = np.
\]
L'approximation est jugée acceptable lorsque $n \geq 30$, $p \leq 0{,}1$ et $\lambda = np \leq 10$.
\end{theoreme}

\begin{exemple}[-- Comparaison chiffrée]
Soit $X \sim \mathcal{B}(200\,;0{,}01)$. Ici $\lambda = np = 2$. On compare la loi exacte et son approximation $\mathcal{P}(2)$ :
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
\rowcolor{uimmlightblue}
$k$ & $0$ & $1$ & $2$ & $3$ & $P(X \leq 3)$ \\
\hline
$\mathcal{B}(200\,;0{,}01)$ & $0{,}1340$ & $0{,}2707$ & $0{,}2720$ & $0{,}1814$ & $0{,}8580$ \\
\hline
$\mathcal{P}(2)$ & $0{,}1353$ & $0{,}2707$ & $0{,}2707$ & $0{,}1804$ & $0{,}8571$ \\
\hline
\end{tabular}
\end{center}
Les écarts sont inférieurs à $2 \times 10^{-3}$ : l'approximation est très satisfaisante.
\end{exemple}

\begin{remarque}[-- Deux approximations à ne pas confondre]
La loi binomiale admet \textbf{deux} approximations selon le contexte :
\begin{itemize}
  \item par la loi de \textbf{Poisson} quand $p$ est \emph{petit} (événements rares) -- ce cours ;
  \item par la loi \textbf{normale} quand $n$ est grand et $p$ \emph{modéré} -- voir le cours « Loi normale ».
\end{itemize}
\end{remarque}

\section{Exercices d'application}

\subsection*{Exercice 1 -- Loi de probabilité \hfill {\normalsize\color{uimmred}[C05]}}
Le nombre $X$ de défauts d'aspect par pièce suit la loi :
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|c|c|}
\hline
\rowcolor{uimmlightblue}
$x_i$ & $0$ & $1$ & $2$ & $3$ \\
\hline
$p_i$ & $0{,}60$ & $0{,}25$ & $?$ & $0{,}05$ \\
\hline
\end{tabular}
\end{center}
\begin{enumerate}[label=\alph*)]
  \item Déterminer $P(X = 2)$.
  \item Calculer $E(X)$ et interpréter.
  \item Calculer l'écart-type $\sigma(X)$.
\end{enumerate}

\subsection*{Exercice 2 -- Loi de Bernoulli \hfill {\normalsize\color{uimmred}[C01]}}
Une soudure est acceptée avec la probabilité $p = 0{,}97$. On note $X = 1$ si la soudure est acceptée, $0$ sinon.
\begin{enumerate}[label=\alph*)]
  \item Donner la loi de $X$.
  \item Calculer $E(X)$ et $\sigma(X)$.
\end{enumerate}

\subsection*{Exercice 3 -- Loi binomiale \hfill {\normalsize\color{uimmred}[C02]\,[C04]}}
Une machine produit $4\,\%$ de pièces non conformes. On prélève un échantillon de $n = 15$ pièces (tirage assimilé à indépendant). Soit $X$ le nombre de pièces non conformes.
\begin{enumerate}[label=\alph*)]
  \item Justifier que $X$ suit une loi binomiale et donner ses paramètres.
  \item Calculer $P(X = 0)$, puis $P(X = 1)$.
  \item Calculer $P(X \leq 2)$ et $E(X)$.
\end{enumerate}

\subsection*{Exercice 4 -- Loi de Poisson \hfill {\normalsize\color{uimmred}[C04]\,[C05]}}
Le nombre de coupures de courant sur une ligne suit une loi de Poisson de moyenne $\lambda = 1{,}5$ par semaine.
\begin{enumerate}[label=\alph*)]
  \item Calculer la probabilité qu'une semaine se passe sans coupure.
  \item Calculer $P(X \geq 2)$.
  \item Donner l'écart-type du nombre hebdomadaire de coupures.
\end{enumerate}

\subsection*{Exercice 5 -- Approximation de Poisson \hfill {\normalsize\color{uimmred}[C04]}}
Un composant électronique est défaillant avec une probabilité $p = 0{,}005$. On en monte $n = 400$ sur une carte. Soit $X$ le nombre de composants défaillants.
\begin{enumerate}[label=\alph*)]
  \item Donner la loi exacte de $X$ et calculer $\lambda = np$.
  \item Vérifier que l'approximation de Poisson est justifiée.
  \item À l'aide de la loi de Poisson, estimer $P(X = 0)$ et $P(X \leq 2)$.
\end{enumerate}

\subsection*{Exercice 6 -- Synthèse : fiabilité d'un système \hfill {\normalsize\color{uimmred}[C02]\,[C04]}}
Un équipement comporte $n = 8$ modules identiques et indépendants. Chaque module fonctionne sans panne sur un an avec la probabilité $0{,}90$. Soit $X$ le nombre de modules encore en service au bout d'un an.
\begin{enumerate}[label=\alph*)]
  \item Justifier que $X$ suit une loi binomiale et donner ses paramètres.
  \item Calculer la probabilité que les $8$ modules tiennent l'année.
  \item Calculer la probabilité qu'au moins $7$ modules soient en service.
  \item Donner le nombre moyen de modules en service au bout d'un an.
\end{enumerate}

\section*{Formulaire -- À retenir}
\addcontentsline{toc}{section}{Formulaire}

\begin{tcolorbox}[colback=uimmlightblue, colframe=uimmblue, title={\textbf{Formulaire des lois discrètes}}, fonttitle=\large\bfseries\sffamily, coltitle=white, boxrule=1.5pt, arc=5pt]

\renewcommand{\arraystretch}{1.9}
\begin{tabular}{|>{\centering\arraybackslash}m{2.6cm}|>{\centering\arraybackslash}m{4.6cm}|>{\centering\arraybackslash}m{2.3cm}|>{\centering\arraybackslash}m{2.8cm}|}
\hline
\rowcolor{uimmblue}
\textcolor{white}{\textbf{Loi}} & \textcolor{white}{\textbf{$P(X=k)$}} & \textcolor{white}{\textbf{$E(X)$}} & \textcolor{white}{\textbf{$\sigma(X)$}} \\
\hline
Bernoulli $\mathcal{B}(p)$ & $P(1)=p,\ P(0)=1-p$ & $p$ & $\sqrt{p(1-p)}$ \\
\hline
Binomiale $\mathcal{B}(n\,;p)$ & $\dbinom{n}{k}p^{k}(1-p)^{n-k}$ & $np$ & $\sqrt{np(1-p)}$ \\
\hline
Poisson $\mathcal{P}(\lambda)$ & $e^{-\lambda}\dfrac{\lambda^{k}}{k!}$ & $\lambda$ & $\sqrt{\lambda}$ \\
\hline
\end{tabular}

\vspace{8pt}
\textbf{Variable discrète générale :} $\displaystyle \sum p_i = 1$ ; \; $E(X) = \sum x_i p_i$ ; \; $V(X) = \sum x_i^2 p_i - E(X)^2$ ; \; $\sigma(X) = \sqrt{V(X)}$.

\vspace{4pt}
\textbf{Coefficient binomial :} $\dbinom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$.

\vspace{4pt}
\textbf{Approximations de $\mathcal{B}(n\,;p)$ :} par $\mathcal{P}(np)$ si $p$ petit ($n\geq30$, $p\leq0{,}1$, $np\leq10$) ; par la loi normale si $n$ grand et $p$ modéré.

\vspace{6pt}
\textbf{Calculatrice.} \; Binomiale -- TI : \texttt{binomFdp/binomFRép} ; Casio : \texttt{Bpd/Bcd}. \; Poisson -- TI : \texttt{poissonFdp/poissonFRép} ; Casio : \texttt{Ppd/Pcd}.
\end{tcolorbox}
